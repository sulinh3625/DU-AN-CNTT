\chapter{PHƯƠNG PHÁP NGHIÊN CỨU VÀ THIẾT KẾ MÔ HÌNH LAI NEUMF-F}
\label{chap:thietke_trienkhai}

Chương này trình bày phương pháp cho kết quả chính của báo cáo: quy trình tổng thể từ dữ liệu đến gợi ý (mục~\ref{sec:system_architecture}), dữ liệu và cách chia giai đoạn -- gồm cả bộ dữ liệu kiểm chứng Amazon Beauty (mục~\ref{sec:hm_pipeline}), mô hình đề xuất NeuMF-F kết hợp nhân tử hóa ma trận với mạng nơ-ron sâu và đặc trưng của sản phẩm, khách hàng, thời gian (mục~\ref{sec:model_design}), cách huấn luyện (mục~\ref{sec:sampling_loss}, \ref{sec:training_pipeline}), quy trình đánh giá không thiên lệch trên một mẫu khách hàng độc lập (mục~\ref{sec:protocol}), cách suy diễn Top-$K$ và xử lý khách hàng mới (mục~\ref{sec:inference}), cùng hệ thống demo (mục~\ref{sec:demo_design}). Các bước xử lý chính được viết thành mã giả (Thuật toán~\ref{alg:kcore}--\ref{alg:inference}). Giai đoạn phát triển trước đó -- chỉ dùng mã khách hàng và mã sản phẩm, đánh giá trên chính mẫu phát triển -- và bản đầu của kế hoạch đánh giá được tóm tắt như lịch sử phát triển ở mục~\ref{sec:v1_history}; những gì rút ra từ hai giai đoạn này là lý do trực tiếp của các lựa chọn trong chương này.

\section{Quy trình tổng thể của hệ thống}
\label{sec:system_architecture}

Hệ thống được tổ chức theo đúng trình tự mà một mô hình gợi ý đi qua khi đưa vào sử dụng (Hình~\ref{fig:system_architecture}): xây dựng đặc trưng $\rightarrow$ huấn luyện $\rightarrow$ lưu mô hình $\rightarrow$ suy diễn. Mỗi lần chạy, đường ống nhận một mốc thời gian $\tau$: mọi bước chỉ dùng dữ liệu trước $\tau$, và bước suy diễn trả về danh sách gợi ý cho giai đoạn sau $\tau$ -- đúng tình huống doanh nghiệp huấn luyện mô hình trên lịch sử đến hôm nay rồi gợi ý cho những tuần tới. Việc đánh giá \emph{không} nằm trong đường ống này mà là một quy trình riêng (mục~\ref{sec:protocol}, Hình~\ref{fig:eval_protocol}): quy trình đó gọi lại đường ống với các mốc khác nhau, rồi mới so danh sách gợi ý với những gì khách đã mua thật.

\subsection{Đặc tả bộ dữ liệu H\&M}
\label{subsec:hm_description}

Bộ dữ liệu H\&M Personalized Fashion Recommendations \parencite{hm2022kaggle} phản ánh hoạt động mua sắm thời trang thực tế của một chuỗi bán lẻ lớn, gồm ba tệp được đề tài sử dụng:
\begin{itemize}
    \item \textbf{\texttt{transactions\_train.csv}} (3,49 GB): 31.788.324 dòng giao dịch từ 20/09/2018 đến 22/09/2020, của 1.362.281 khách hàng trên 104.547 sản phẩm. Ba trường được dùng: \texttt{customer\_id} (mã khách, chuỗi băm 64 ký tự), \texttt{article\_id} (mã sản phẩm, 10 chữ số) và \texttt{t\_dat} (ngày mua, không có giờ). Giá và kênh bán không được dùng.
    \item \textbf{\texttt{articles.csv}}: 105.542 sản phẩm của danh mục, mỗi sản phẩm có các thuộc tính dạng mã (loại sản phẩm, nhóm sản phẩm, họa tiết, nhóm màu, độ đậm màu, màu chủ đạo, bộ phận, chỉ mục, nhóm chỉ mục, khu, nhóm may mặc) và mô tả văn bản (tên, mô tả chi tiết).
    \item \textbf{\texttt{customers.csv}}: 1.371.980 khách hàng với tuổi, trạng thái hội viên, tần suất nhận tin thời trang và hai cờ FN, Active. Mã bưu chính không được dùng. Tệp này có nhiều dòng hơn giới hạn của bảng tính Excel (1.048.576 dòng), nên phải được đọc trực tiếp bằng chương trình; hệ thống kiểm tra số dòng trước khi dùng.
\end{itemize}
Ảnh sản phẩm có trong bộ dữ liệu nhưng không được dùng làm đầu vào của mô hình; ảnh chỉ được hiển thị trong hệ thống demo (mục~\ref{sec:demo_design}).

Đánh giá Full Ranking (mục~\ref{sec:metrics}) phải chấm điểm \emph{mọi} sản phẩm ứng viên cho mỗi người dùng được đánh giá, nên chi phí tỉ lệ với tích số người dùng và số sản phẩm. Bảng~\ref{tab:kcore_hm_full} cho thấy ở quy mô đầy đủ, ngay cả với $k=10$ vẫn còn 633.130 người dùng và 80.265 sản phẩm -- tức hàng chục tỷ cặp cần tính điểm cho mỗi mô hình, vượt xa khả năng của phần cứng sẵn có (máy tính cá nhân, RAM 16 GB). Vì vậy đề tài thực nghiệm trên các mẫu theo khách hàng (mục~\ref{subsec:hm_sampling}).

\subsection{Lấy mẫu theo khách hàng: mẫu phát triển và mẫu kiểm định}
\label{subsec:hm_sampling}

Đề tài lấy mẫu \emph{theo khách hàng} thay vì theo dòng: lấy mẫu theo dòng (hoặc cắt $n$ dòng đầu) sẽ cắt vụn lịch sử của từng khách và -- vì tệp gốc sắp theo ngày -- chỉ phủ vài ngày giao dịch, làm hỏng cả bước lọc $k$-core lẫn việc chia dữ liệu theo thời gian. Quy trình gồm hai lượt đọc tệp theo lô (bộ nhớ đỉnh khoảng 0,5 GB):
\begin{enumerate}
    \item \textbf{Lượt 1 -- chọn khách:} băm \texttt{customer\_id} bằng một khoá cố định vào 100.000 nhóm (hàm băm phân khách vào nhóm một cách giả ngẫu nhiên, độc lập với hành vi mua) và đếm số giao dịch của từng nhóm.
    \item \textbf{Lượt 2 -- ghi mẫu:} giữ \emph{toàn bộ} giao dịch của các khách được chọn, trải đủ 20/09/2018 -- 22/09/2020.
\end{enumerate}
Các mẫu được lấy từ những đoạn nhóm băm liền nhau và không chồng lên nhau (gọi là các \emph{khối}); mỗi khối gồm số nhóm ít nhất để đủ khoảng 500.000 giao dịch:
\begin{itemize}
    \item \textbf{Mẫu phát triển} -- khối 0, các nhóm $[0, 1597)$: 500.269 giao dịch của 21.599 khách hàng. Mẫu này được dùng trong toàn bộ giai đoạn phát triển ban đầu và, ở bản cuối của quy trình, chỉ dùng để tinh chỉnh trên tập xác thực.
    \item \textbf{Khối 1} -- các nhóm $[1597, 3160)$: 500.125 giao dịch của 21.030 khách hàng. Khối này là mẫu kiểm định của bản đầu kế hoạch đánh giá và đã được mở ở lần đánh giá cuối ngày 04/10/2026, nên không còn dùng để kết luận (mục~\ref{sec:v1_history}).
    \item \textbf{Mẫu kiểm định} -- khối 2, các nhóm $[3160, 4734)$: 500.164 giao dịch của 21.351 khách hàng. Vì các đoạn nhóm không chồng nhau, \textbf{không có khách hàng nào thuộc cả mẫu phát triển lẫn mẫu kiểm định} (đã kiểm tra trực tiếp). Mẫu kiểm định chỉ dùng cho đánh giá cuối theo bản cuối của kế hoạch.
\end{itemize}
Vì dựa trên hàm băm với khoá cố định, chạy lại luôn cho đúng cùng các tệp (mã băm MD5 của hai tệp mẫu được ghi trong kế hoạch đăng ký trước, mục~\ref{sec:protocol}). Lý do cần một mẫu kiểm định riêng: tập kiểm thử của mẫu phát triển đã được xem nhiều lần trong giai đoạn phát triển, và tập kiểm thử của khối 1 đã được mở trước khi kế hoạch được sửa thành bản cuối (mục~\ref{sec:v1_history}), nên cả hai không còn là dữ liệu ``chưa từng thấy''. Mẫu kiểm định gồm những khách hàng khác hẳn, nên kết quả trên mẫu này ước lượng hiệu năng của mô hình khi gặp khách hàng khác của cùng cửa hàng, trong cùng giai đoạn thời gian.

\subsection{Ma trận tương tác và lọc \texorpdfstring{$k$}{k}-core trước mốc kiểm thử}
\label{subsec:interaction_matrix}

\textbf{Một sản phẩm là một mẫu thiết kế.} Mã \texttt{article\_id} 10 chữ số của H\&M gồm mã mẫu thiết kế \texttt{product\_code} và ba chữ số cuối chỉ biến thể màu, nên cùng một mẫu áo có thể mang nhiều \texttt{article\_id}. Theo góp ý của giảng viên hướng dẫn, bản cuối của giao thức coi một sản phẩm là một \texttt{product\_code}: mọi màu của cùng một mẫu được gộp làm một sản phẩm, và khách mua nhiều màu của cùng một mẫu chỉ được tính một cặp (khách, sản phẩm). Trước mốc kiểm thử, việc gộp này giảm số sản phẩm của mẫu phát triển từ 55.971 xuống 27.370 và số cặp (khách, sản phẩm) từ 401.293 xuống 359.181; với mẫu kiểm định là từ 55.683 xuống 27.291 sản phẩm và từ 398.653 xuống 356.740 cặp.

Một khách có thể mua cùng một sản phẩm nhiều lần; các giao dịch lặp được gộp thành một cặp (khách, sản phẩm) duy nhất, giữ lại \emph{ngày mua đầu tiên} $t_{ui}$, \emph{trước} khi lọc $k$-core. Nếu lọc trực tiếp trên dòng giao dịch thô, bậc của một khách/sản phẩm bị thổi phồng bởi các lượt mua lặp: trên mẫu phát triển (theo \texttt{article\_id}, cả hai năm), 500.269 giao dịch gộp lại chỉ còn 429.964 cặp duy nhất.

Lọc $k$-core được áp dụng lặp đến khi hội tụ (loại một sản phẩm có thể khiến một khách rơi xuống dưới ngưỡng và ngược lại), để mỗi khách và mỗi sản phẩm còn lại có ít nhất $k$ đối tác riêng biệt. Điểm khác với giai đoạn phát triển ban đầu là phép lọc \textbf{chỉ dùng các cặp có ngày mua đầu trước mốc kiểm thử} (29/07/2020): việc chọn khách và sản phẩm nào được giữ lại không dùng bất kỳ thông tin nào của giai đoạn kiểm thử. Thuật toán~\ref{alg:kcore} tóm tắt toàn bộ bước chuẩn bị dữ liệu. Sau khi lọc với $k = 10$, mẫu phát triển còn \VData{A}{nusers} khách, \VData{A}{nitems} sản phẩm và \VData{A}{nkeptpairs} cặp; mẫu kiểm định còn \VData{B}{nusers} khách, \VData{B}{nitems} sản phẩm và \VData{B}{nkeptpairs} cặp. Mã khách và mã sản phẩm được đánh lại chỉ số liên tục, và phản hồi ẩn nhị phân $y_{ui} = 1$ khi khách $u$ đã mua sản phẩm $i$ (Công thức~\eqref{eq:implicit_binary}). Dữ liệu đã lọc của mỗi mẫu được ghi ra một tệp, kèm mã MD5 của nội dung; mọi mô hình đọc cùng tệp này và chương trình kiểm tra mã MD5 trước khi dùng, nên mọi mô hình chắc chắn được huấn luyện và đánh giá trên cùng dữ liệu.

\begin{algorithm}[htbp]
\caption{Chuẩn bị dữ liệu: gộp theo mẫu thiết kế và lọc $k$-core lặp}
\label{alg:kcore}
\begin{algorithmic}[1]
\Require giao dịch $(c, a, t)$ của một mẫu (khách $c$, mã \texttt{article\_id} $a$, ngày mua $t$); mốc kiểm thử $\tau_{test}$; ngưỡng $k$
\Ensure tệp cặp $(c, i, t_{ci}, \textit{thuộc\_kcore})$ và mã MD5 của nội dung tệp
\For{mỗi giao dịch $(c, a, t)$}
    \State $i \gets \lfloor a / 1000 \rfloor$ \Comment{\texttt{product\_code}: bỏ 3 chữ số chỉ màu}
\EndFor
\State gộp các giao dịch theo cặp $(c, i)$, giữ $t_{ci} \gets$ ngày mua sớm nhất \Comment{mỗi cặp một dòng}
\State $E \gets \{(c, i) : t_{ci} < \tau_{test}\}$ \Comment{chỉ dữ liệu trước mốc kiểm thử}
\Repeat
    \State $U \gets \{c : |\{i : (c, i) \in E\}| \ge k\}$; \quad $I \gets \{i : |\{c : (c, i) \in E\}| \ge k\}$
    \State $E' \gets \{(c, i) \in E : c \in U,\ i \in I\}$; \quad $\textit{hội\_tụ} \gets (|E'| = |E|)$; \quad $E \gets E'$
\Until{\textit{hội\_tụ}}
\State ghi \emph{mọi} cặp $(c, i, t_{ci})$ kèm cờ $(c, i) \in E$ ra tệp CSV nén, theo thứ tự cố định
\State $\textit{md5} \gets \mathrm{MD5}(\text{nội dung tệp})$ \Comment{ghi vào \texttt{manifest.json}}
\State \Return tệp, \textit{md5}
\end{algorithmic}
\end{algorithm}

Khi nạp dữ liệu cho một giai đoạn, chương trình kiểm tra lại mã MD5, giữ các khách và sản phẩm có cờ thuộc $k$-core, rồi lấy dữ liệu huấn luyện và sản phẩm đúng theo các tập ở mục~\ref{subsec:split_sampling}; vì tệp chứa mọi cặp (kể cả sau mốc kiểm thử), tập kiểm thử chỉ được dựng khi chạy đánh giá cuối.

\label{subsec:hm_apply}

\subsection{Giai đoạn đánh giá và tập ứng viên}
\label{subsec:split_sampling}

Dữ liệu được chia theo thời gian bằng hai mốc cố định chung cho mọi khách hàng: mốc xác thực $\tau_{val}$ = 01/07/2020 (cửa sổ xác thực 01/07 -- 28/07/2020, 4 tuần) và mốc kiểm thử $\tau_{test}$ = 29/07/2020 (cửa sổ kiểm thử 29/07 -- 22/09/2020, 8 tuần). Với mỗi mốc $\tau$ (gọi là một \emph{giai đoạn}), các tập sau được định nghĩa:
\begin{align}
    \mathcal{D}_\tau &= \{(u, i) \text{ đã lọc } k\text{-core} : t_{ui} < \tau\} && \text{(dữ liệu huấn luyện)} \label{eq:v2_train}\\
    \mathcal{I}_\tau &= \{i : \exists u,\ (u, i) \in \mathcal{D}_\tau\} && \text{(sản phẩm có dữ liệu huấn luyện)} \label{eq:v2_old}\\
    \mathcal{C}_u(\tau) &= \mathcal{I}_\tau \setminus \mathcal{I}_u(\tau) && \text{(tập ứng viên của khách } u) \label{eq:v2_cand}\\
    T_u(\tau) &= \{i \in \mathcal{C}_u(\tau) : \tau \le t_{ui} < \tau_{end}\} && \text{(sản phẩm đúng của } u) \label{eq:v2_target}
\end{align}
trong đó $\mathcal{I}_u(\tau)$ là các sản phẩm $u$ đã mua trước $\tau$, và $\tau_{end}$ là cuối cửa sổ của giai đoạn. Chỉ những khách có dữ liệu huấn luyện trong $\mathcal{D}_\tau$ và có ít nhất một sản phẩm đúng mới được đánh giá. Hai điểm cần giải thích:
\begin{itemize}
    \item \textbf{Chỉ sản phẩm đã có dữ liệu huấn luyện.} Sản phẩm chưa có cặp nào trong dữ liệu huấn luyện của giai đoạn -- trong đó có sản phẩm vừa ra mắt -- không thuộc tập ứng viên lẫn tập sản phẩm đúng. Đây là thay đổi của bản cuối kế hoạch đánh giá, theo góp ý của giảng viên hướng dẫn (bản đầu của kế hoạch giữ sản phẩm mới trong tập ứng viên, mục~\ref{sec:v1_history}). Với tập ứng viên này, mọi mô hình -- kể cả mô hình chỉ dùng ID -- đều chấm được mọi ứng viên, nên so sánh giữa các mô hình không phụ thuộc vào khả năng xử lý sản phẩm chưa có lịch sử bán; đổi lại, kết quả không nói về khả năng gợi ý sản phẩm vừa ra mắt (mục~\ref{sec:pham_vi_thuc_nghiem}).
    \item \textbf{Huấn luyện lại trên dữ liệu mới nhất.} Ở giai đoạn kiểm thử, $\mathcal{D}_{\tau_{test}}$ gồm cả cửa sổ xác thực, tức ``tập huấn luyện $\cup$ xác thực''. Tập xác thực chỉ dùng để chọn cấu hình và số epoch; trước khi đánh giá trên tập kiểm thử, mô hình được huấn luyện lại từ đầu trên $\mathcal{D}_{\tau_{test}}$ (mục~\ref{sec:protocol}), nên khi dự đoán, mô hình đã thấy toàn bộ quá khứ như lúc triển khai thật.
\end{itemize}
\paragraph{Vì sao không chia theo Leave-One-Out.} Cách chia của \textcite{he2017ncf} lấy sản phẩm mua cuối cùng của từng người dùng làm tập kiểm thử, phần còn lại làm tập huấn luyện. Cách này để lọt thông tin tương lai: tập huấn luyện chứa tương tác của những người dùng khác xảy ra \emph{sau} thời điểm cần dự đoán \parencite{meng2020splitting}. Đo trên mẫu phát triển ở giai đoạn phát triển, tính trung bình theo người dùng, 11,2\% tương tác trong tập huấn luyện xảy ra sau ngày mua sản phẩm kiểm thử của người dùng đó; ngoài ra 60,9\% người dùng có sản phẩm xác thực và sản phẩm kiểm thử mua cùng một ngày. Vì vậy đề tài chia theo mốc thời gian chung cho mọi người dùng.

Theo định nghĩa, mỗi cặp (khách, sản phẩm) chỉ có một ngày mua đầu, nên một cặp không thể vừa là dữ liệu huấn luyện vừa là sản phẩm đúng của cùng giai đoạn, và sản phẩm khách đã mua trước mốc không bao giờ là sản phẩm đúng -- mô hình dự đoán những món \emph{mới đối với khách}. Các điều kiện này, cùng việc lọc $k$-core chỉ dùng dữ liệu trước mốc kiểm thử, được kiểm chứng bằng bộ kiểm thử tự động. \bangrefV{tab_v2_data.tex}{tab:v2data} tổng hợp quy mô dữ liệu của hai mẫu.

\paragraph{Đặc trưng thời gian của sản phẩm.} Gọi $n_{i,d}$ là số giao dịch của sản phẩm $i$ trong ngày $d$ trên toàn bộ H\&M và $d^0_i$ là ngày bán đầu tiên. Tại ngày $t$:
\begin{equation}
    \begin{aligned}
        s^{(w)}_i(t) &= \sum_{d = t - w}^{t - 1} n_{i,d}, \\
        x^{tg}_i(t) &= \Big[\tfrac{\log(1 + s^{(7)}_i)}{10},\ \tfrac{\log(1 + s^{(28)}_i)}{10},\ \tfrac{\log(1 + s^{(91)}_i)}{10},\ \tfrac{\log(1 + \max(0,\, t - d^0_i))}{10},\ \mathbb{I}(d^0_i \ge t)\Big]
    \end{aligned}
    \label{eq:item_time}
\end{equation}
Tổng chỉ chạy tới ngày $t-1$, nên doanh số của chính ngày $t$ không được dùng. Phép $\log(1+x)$ nén các sản phẩm bán rất chạy, hệ số $1/10$ đưa đặc trưng về cỡ $[0, 1]$. Doanh số lấy trên toàn bộ giao dịch H\&M chứ không chỉ trong mẫu -- đây là dữ liệu mà doanh nghiệp có tại thời điểm dự đoán, và phản ánh xu hướng bán gần đây của sản phẩm trên toàn cửa hàng.

\paragraph{Đặc trưng văn bản.} Tên và mô tả của mỗi sản phẩm được biểu diễn bằng TF-IDF \parencite{salton1988term} trên từ đơn và cặp từ (bỏ từ dừng tiếng Anh, giữ tối đa 30.000 mục từ xuất hiện ở ít nhất 3 sản phẩm, tần suất từ theo thang logarit), rồi giảm còn 64 chiều bằng phân rã SVD rút gọn -- tức phân tích ngữ nghĩa tiềm ẩn \parencite{deerwester1990lsa} -- và chuẩn hóa về độ dài 1. Phép biến đổi này được khớp trên toàn bộ danh mục và không dùng dữ liệu mua hàng.

\paragraph{Gộp đặc trưng theo mẫu thiết kế.} Đặc trưng ban đầu được tính cho từng \texttt{article\_id}; vì một sản phẩm là một \texttt{product\_code} (mục~\ref{subsec:interaction_matrix}), đặc trưng của các biến thể màu được gộp lại: thuộc tính danh mục lấy theo biến thể có \texttt{article\_id} nhỏ nhất, ba thuộc tính màu (nhóm màu, độ đậm màu, màu chủ đạo) đặt về một giá trị hằng vì không còn mô tả một màu cụ thể, vector văn bản là trung bình các biến thể rồi chuẩn hóa lại về độ dài 1, doanh số theo ngày là tổng các biến thể, và ngày bán đầu tiên là ngày sớm nhất của các biến thể.

\paragraph{Đặc trưng của khách hàng.} Thuộc tính tĩnh lấy từ \texttt{customers.csv}; tệp này không có mốc thời gian nên thuộc tính được coi là ít thay đổi (một hạn chế được ghi nhận, mục~\ref{sec:hanche}). Đặc trưng thời gian của khách tại ngày $t$ chỉ dùng các cặp có ngày mua đầu \emph{trước} $t$: khoảng cách tới lần mua gần nhất và số cặp đã mua.

Bộ kiểm thử tự động xác nhận rằng đặc trưng thời gian của sản phẩm tại ngày $t$ không thay đổi khi thêm giao dịch của ngày $t$ trở về sau, và đặc trưng của khách tại ngày $t$ chỉ đếm các cặp trước $t$.

\subsection{Dữ liệu kiểm chứng Amazon Beauty}
\label{subsec:amazon_data}

Để xem các kết luận rút ra từ H\&M có còn đúng trên một bộ dữ liệu khác hay không, toàn bộ quy trình được chạy lại trên \textbf{Amazon Beauty} -- danh mục Làm đẹp của bộ dữ liệu đánh giá sản phẩm Amazon năm 2014 \parencite{mcauley2015image}, bản 5-core: 198.502 đánh giá của 22.363 khách cho 12.101 sản phẩm, từ 12/06/2002 đến 23/07/2014. Kế hoạch kiểm chứng được đăng ký trước khi chạy bất kỳ mô hình nào trên bộ này và trước khi mở tập kiểm thử của nó. Mã nguồn, 14 mô hình, số seed, độ đo, họ 10 so sánh và tiêu chí kết luận giữ nguyên như H\&M; chỉ những điểm sau thay đổi theo dữ liệu:
\begin{itemize}
    \item \textbf{Tương tác:} mỗi đánh giá được coi là một tương tác ẩn giữa khách và sản phẩm (như một lượt mua của H\&M), không dùng số sao. Một sản phẩm là một mã \texttt{asin}, không gộp biến thể.
    \item \textbf{Chia theo thời gian:} chọn số epoch bằng các cặp trước 01/10/2013 và tập xác thực 01/10 -- 31/12/2013; huấn luyện lại trên mọi cặp trước 01/01/2014; tập kiểm thử 01/01 -- 23/07/2014 (tỉ lệ độ dài xác thực : kiểm thử khoảng 1 : 2 như H\&M). Vì chỉ có một bộ dữ liệu, Amazon \emph{không có mẫu khách hàng độc lập} như H\&M: bước chọn số epoch và bước đánh giá cuối khác nhau theo thời gian chứ không theo khách hàng.
    \item \textbf{Lọc 5-core} chỉ trên các cặp trước mốc kiểm thử -- bằng ngưỡng của bản 5-core gốc; H\&M dùng $k = 10$ nhưng dữ liệu Amazon thưa hơn nhiều.
    \item \textbf{Không tinh chỉnh lại:} mọi mô hình dùng đúng cấu hình đã chọn trên tập xác thực của mẫu phát triển H\&M (\bangrefV{tab_v2_tuning.tex}{tab:v2tuning}); thứ duy nhất được chọn trên dữ liệu Amazon là số epoch của các mạng nơ-ron, theo đúng Thuật toán~\ref{alg:train}. Hệ quả: mô hình nào có cấu hình của H\&M không hợp với Amazon sẽ chịu thiệt -- và không tách được phần thiệt đó khỏi bản thân mô hình (mục~\ref{sec:amazon_results}).
    \item \textbf{Đặc trưng} thay theo những gì bộ dữ liệu có (Bảng~\ref{tab:amazon_features}). Amazon không có thông tin khách hàng, nên nhánh khách của NeuMF-F chỉ còn hai đặc trưng thời gian; doanh số được thay bằng số đánh giá, đếm trên chính tệp 5-core.
\end{itemize}

\section{Thiết kế mô hình lai NeuMF-F}
\label{sec:model_design}

NeuMF-F giữ nguyên cấu trúc hai nhánh của NeuMF \parencite{he2017ncf} -- nhánh GMF mô hình hóa tương tác tuyến tính, nhánh MLP mô hình hóa tương tác phi tuyến, hợp nhất ở lớp dự đoán chung -- và thay đổi một điểm: vector của khách hàng và sản phẩm trong mỗi nhánh không còn chỉ là Embedding của mã ID mà là \emph{tổng của Embedding ID và một phép chiếu tuyến tính của đặc trưng}. Chữ ``F'' (features) chỉ phần đặc trưng này. \subsection{Biểu diễn lai: Embedding ID kết hợp đặc trưng}
\label{subsec:embedding_design}

Với mỗi nhánh $b \in \{G, M\}$ (GMF và MLP), mô hình có bảng Embedding ID của khách hàng $P^b \in \mathbb{R}^{|\mathcal{U}| \times d}$, của sản phẩm $Q^b \in \mathbb{R}^{|\mathcal{I}^{ID}| \times d}$ và hai ma trận chiếu đặc trưng $W^b_U \in \mathbb{R}^{d \times d_U}$, $W^b_I \in \mathbb{R}^{d \times d_I}$ (không có hệ số chệch). Vector của khách $u$ và sản phẩm $i$ tại ngày $t$ trong nhánh $b$ là:
\begin{equation}
    p^b_u(t) = P^b_u + W^b_U\, x_u(t), \qquad
    q^b_i(t) = m_i\, Q^b_i + W^b_I\, x_i(t)
    \label{eq:neumff_embedding}
\end{equation}
trong đó vector đặc trưng của sản phẩm $x_i(t) = [e(a_{i,1}); \dots; e(a_{i,11});\ v_i;\ x^{tg}_i(t)] \in \mathbb{R}^{d_I}$ gồm Embedding 8 chiều của 11 thuộc tính danh mục, vector văn bản $v_i$ (64 chiều) và 5 đặc trưng thời gian (Công thức~\eqref{eq:item_time}), tức $d_I = 88 + 64 + 5 = 157$; tương tự $x_u(t) = [e(c_{u,1}); \dots; e(c_{u,5});\ x^{tg}_u(t)] \in \mathbb{R}^{d_U}$ với $d_U = 40 + 2 = 42$. Embedding của các thuộc tính $e(\cdot)$ được học cùng mô hình và dùng chung cho hai nhánh; ma trận chiếu $W^b$ riêng cho từng nhánh.

Hệ số $m_i$ quyết định Embedding ID của sản phẩm có được dùng hay không: $m_i = 1$ với sản phẩm có dữ liệu huấn luyện ở giai đoạn đang xét -- với tập ứng viên của đề tài (mục~\ref{subsec:split_sampling}), đó là mọi sản phẩm được chấm -- còn khi $m_i = 0$, vector của sản phẩm chỉ còn phần đặc trưng. Khi huấn luyện, $m_i$ của từng mẫu được đặt về 0 ngẫu nhiên với xác suất $\rho$ (``bỏ ID ngẫu nhiên'', $\rho \in \{0;\, 0{,}25;\, 0{,}5\}$ được tinh chỉnh). Cơ chế này buộc mô hình học cách chấm điểm một phần từ đặc trưng thay vì dựa hoàn toàn vào Embedding ID; ý tưởng bỏ ngẫu nhiên đầu vào sở thích tham khảo từ DropoutNet \parencite{volkovs2017dropoutnet}, vốn được đề xuất để chấm điểm đối tượng khởi đầu lạnh. Trong đề tài, $\rho$ là một siêu tham số như các siêu tham số khác và được chọn trên tập xác thực; cấu hình NeuMF-F được chọn có $\rho = 0$, tức không bỏ ID (mục~\ref{sec:tuning_results}).

Như NeuMF, hai nhánh dùng hai không gian nhúng độc lập: nhánh GMF tối ưu vector cho phép nhân Hadamard (tuyến tính), nhánh MLP tối ưu vector làm đầu vào của mạng phi tuyến sâu; ép hai mục tiêu khác bản chất này dùng chung tham số có thể khiến gradient của hai nhánh xung đột \parencite{he2017ncf}. Hai nhánh dùng cùng số chiều $d \in \{32, 64\}$. Bảng Embedding và Embedding thuộc tính được khởi tạo theo phân phối chuẩn $\mathcal{N}(0;\, 0{,}01^2)$ như công trình gốc, hàng ``không có ID'' là vector 0 cố định; các tầng tuyến tính khởi tạo theo Xavier; lớp đầu ra $h$ không có hệ số chệch.

\subsection{Nhánh GMF-F}
\label{subsec:gmf_design}

Nhánh GMF-F là dạng tổng quát hóa của nhân tử hóa ma trận (Công thức~\eqref{eq:mf_dot}) trên vector lai: thay vì cộng dồn ngay tích các chiều thành một số vô hướng, nhánh giữ kết quả nhân theo từng chiều qua phép nhân Hadamard:
\begin{equation}
    \phi^{G} = p^G_u(t) \odot q^G_i(t)
    \label{eq:gmff}
\end{equation}
Phép cộng dồn được chuyển sang lớp trọng số $h$ ở tầng đầu ra (mục~\ref{subsec:neumf_output}), cho phép mô hình học trọng số riêng cho từng chiều -- phần ``tổng quát hóa'' trong tên gọi GMF; khi $h = \mathbf{1}$ và không có đặc trưng, GMF-F trở về MF. Vì $p^G_u$ và $q^G_i$ đều chứa phần chiếu của đặc trưng, tích $p^{G}_u \odot q^{G}_i$ còn mô hình hóa tương tác tuyến tính giữa thuộc tính của khách và thuộc tính của sản phẩm -- tinh thần tương tự Factorization Machines \parencite{rendle2010fm}, mô hình tổng quát hóa MF cho vector đặc trưng bất kỳ.

\subsection{Nhánh MLP-F}
\label{subsec:mlp_design}

Nhánh MLP-F \emph{nối} hai vector lai thành đầu vào của một mạng nhiều tầng, không áp đặt trước cách hai vector tương tác:
\begin{equation}
    z_1 = \begin{bmatrix} p^M_u(t) \\ q^M_i(t) \end{bmatrix} \in \mathbb{R}^{2d}, \qquad
    \phi^M_L = a_L\big(W_L^T \cdots a_2(W_2^T z_1 + b_2) \cdots + b_L\big)
    \label{eq:mlpf}
\end{equation}
Tháp có kiến trúc giảm một nửa qua mỗi tầng ẩn theo thiết kế của \textcite{he2017ncf}: $[2d \to d \to d/2 \to d/4]$ (với $d = 32$: $[64 \to 32 \to 16 \to 8]$). Mỗi tầng ẩn gồm một phép biến đổi tuyến tính, hàm kích hoạt ReLU và Dropout (tỉ lệ tinh chỉnh trong $\{0;\, 0{,}2\}$) để hạn chế quá khớp. Cùng với Dropout, mô hình được chống quá khớp bằng điều chuẩn $L_2$ (mục~\ref{subsec:bce_regularization}) và dừng sớm theo NDCG@10 trên tập xác thực (mục~\ref{subsec:hyperparameters_setup}).

\subsection{Tầng hợp nhất và lớp dự đoán đầu ra}
\label{subsec:neumf_output}

Vector của nhánh GMF-F (tương tác tuyến tính) và của nhánh MLP-F (tương tác phi tuyến) được nối thành một vector hợp nhất rồi qua một tầng tuyến tính cuối và hàm Sigmoid:
\begin{equation}
    \hat{y}_{ui} = \sigma\big(h^T [\phi^{G};\ \phi^M_L]\big) = \sigma\big(h_G^T\, (p^G_u \odot q^G_i) + h_M^T\, \phi^M_L\big)
    \label{eq:neumff_predict}
\end{equation}
Theo phân loại ở mục~\ref{subsec:early_late_fusion}, đây là \textbf{hợp nhất sớm (Early Fusion) ở mức biểu diễn} của view tuyến tính (MF) và view phi tuyến (DNN): một lớp dự đoán chung học trọng số giữa hai view, và toàn bộ mạng được huấn luyện chung từ đầu đến cuối. Đồng thời, đặc trưng của sản phẩm và khách hàng được đưa vào ngay ở mức vector của cả hai nhánh, tức thông tin nội dung và ngữ cảnh cũng được hợp nhất sớm với thông tin cộng tác (Embedding ID). Khi xếp hạng, hệ thống dùng trực tiếp giá trị trước Sigmoid (logit): Sigmoid đơn điệu nên không đổi thứ tự, nhưng ở độ chính xác float32 nó bão hòa về đúng 1,0 với điểm lớn và tạo ra các giá trị bằng nhau giả. Khi đánh giá ở một giai đoạn, vector của mọi khách và mọi sản phẩm tại mốc $\tau$ được tính một lần rồi dùng lại cho mọi cặp, nên chi phí chấm toàn bộ tập ứng viên gần với chi phí của NeuMF chỉ dùng ID.

\subsection{Các mô hình đối chiếu xây dựng trên cùng khung}
\label{subsec:early_fusion_design}

Để trả lời từng câu hỏi nghiên cứu bằng một phép so sánh trực tiếp, các mô hình sau được cài đặt trên cùng mã nguồn, cùng đặc trưng và cùng quy trình huấn luyện với NeuMF-F:
\begin{itemize}
    \item \textbf{GMF-F} và \textbf{MLP-F}: NeuMF-F chỉ giữ một nhánh -- đo đóng góp của từng nhánh và lợi ích của việc hợp nhất. MLP-F đứng riêng là một mô hình DNN thuần (chỉ có view phi tuyến), không phải mô hình hợp nhất MF với DNN.
    \item \textbf{LateFusion-F}: hợp nhất muộn (Late Fusion) của cùng hai thành phần -- GMF-F và MLP-F được huấn luyện \emph{riêng}, rồi trộn điểm đã chuẩn hóa min--max trên tập ứng viên của từng khách theo trọng số $w$ chọn trên tập xác thực (Công thức~\eqref{eq:late_fusion_mfdnn}). So sánh NeuMF-F với LateFusion-F là so sánh hai cách hợp nhất trên cùng hai thành phần.
    \item \textbf{GMF, MLP, NeuMF chỉ dùng ID}: kiến trúc NCF gốc \parencite{he2017ncf}, không có đặc trưng -- đo lợi ích của đặc trưng.
    \item \textbf{Ablation của NeuMF-F}: cùng siêu tham số với NeuMF-F, tắt lần lượt một thành phần -- vector văn bản, đặc trưng thời gian, thông tin khách hàng, thuộc tính sản phẩm, hoặc cơ chế bỏ ID ngẫu nhiên. Biến thể nào trùng với chính NeuMF-F thì không chạy (mục~\ref{subsec:ablation}).
\end{itemize}
Các baseline không thuộc khung này (Most Popular, MostPopular-Recent, Content, ItemKNN, UserKNN, BPR-MF) và mô hình lai truyền thống Hybrid-CBCF -- kết hợp Content, UserKNN và ItemKNN, bổ sung sau đánh giá cuối (Thuật toán~\ref{alg:hybrid}) -- được mô tả ở mục~\ref{sec:baselines}, cùng cách xếp mọi mô hình đối chứng thành ba nhóm.

\section{Lấy mẫu âm theo thời điểm và hàm mục tiêu}
\label{sec:sampling_loss}

\subsection{Lấy mẫu âm động theo ngày ra mắt của sản phẩm}
\label{subsec:dynamic_sampling}

Dữ liệu phản hồi ẩn (mục~\ref{subsec:feedback_types}) chỉ ghi nhận tương tác đã xảy ra, không có nhãn âm tường minh; Negative Sampling giả định các sản phẩm khách chưa mua nhiều khả năng là âm \parencite{he2017ncf}. Với mô hình có đặc trưng, mỗi cặp dương $(u, i)$ mang ngày mua đầu $t_{ui}$, nên việc chọn mẫu âm phải nhất quán về thời gian: với mỗi cặp dương, hệ thống lấy ngẫu nhiên $N_{ratio}$ sản phẩm $j$ trong các sản phẩm của tập huấn luyện \textbf{đã ra mắt trước hoặc trong ngày $t_{ui}$} ($d^0_j \le t_{ui}$), trừ những sản phẩm khách $u$ đã mua, và gán nhãn $y_{uj} = 0$; mẫu âm được tính đặc trưng tại cùng ngày $t_{ui}$. Một sản phẩm chưa ra mắt ở ngày $t_{ui}$ thì khách không thể mua, nên không phải mẫu âm có ý nghĩa; nếu vẫn lấy, mô hình có thể học quy luật sai ``sản phẩm có cờ chưa ra mắt là âm'' -- một tín hiệu chỉ có trong cách tạo dữ liệu huấn luyện, không phản ánh sở thích của khách. Với mô hình chỉ dùng ID, mẫu âm được lấy đều trong các sản phẩm có ID, trừ sản phẩm khách đã mua, như NCF gốc. $N_{ratio}$ được tinh chỉnh trong $\{4, 8\}$ (tham khảo thiết lập 3--6 mẫu âm cho mỗi mẫu dương của \textcite{he2017ncf}); mẫu âm được lấy lại ở \emph{mỗi} epoch, giúp mô hình thấy nhiều tổ hợp cặp âm khác nhau. Thuật toán~\ref{alg:negative} mô tả bước này cho mô hình có đặc trưng.

\begin{algorithm}[htbp]
\caption{Lấy mẫu âm theo thời điểm cho mô hình có đặc trưng (thực hiện lại ở đầu mỗi epoch)}
\label{alg:negative}
\begin{algorithmic}[1]
\Require tập cặp dương huấn luyện $\mathcal{D} = \{(u, i, t_{ui})\}$; ngày bán đầu tiên $d^0_j$ của mỗi sản phẩm; số mẫu âm $N_{ratio}$
\Ensure tập huấn luyện của epoch: các bộ $(u, j, t, y)$
\State $\mathcal{P} \gets$ các sản phẩm xuất hiện trong $\mathcal{D}$, sắp tăng dần theo $d^0_j$
\State $\mathcal{I}_u \gets \{i : (u, i, \cdot) \in \mathcal{D}\}$ với mọi khách $u$ \Comment{sản phẩm khách đã mua}
\State $\mathcal{B} \gets \emptyset$
\For{mỗi $(u, i, t) \in \mathcal{D}$}
    \State thêm $(u, i, t, 1)$ vào $\mathcal{B}$ \Comment{mẫu dương, đặc trưng tính tại ngày $t$}
    \State $\mathcal{A}_t \gets \{j \in \mathcal{P} : d^0_j \le t\}$ \Comment{sản phẩm đã ra mắt tính đến ngày $t$}
    \For{$n = 1, \dots, N_{ratio}$}
        \State lấy $j$ ngẫu nhiên đều trong $\mathcal{A}_t$; nếu $j \in \mathcal{I}_u$ thì lấy lại (tối đa 20 lần)
        \State thêm $(u, j, t, 0)$ vào $\mathcal{B}$ \Comment{mẫu âm dùng cùng ngày $t$}
    \EndFor
\EndFor
\State \Return $\mathcal{B}$
\end{algorithmic}
\end{algorithm}

\subsection{Hàm tổn thất Binary Cross-Entropy và điều chuẩn}
\label{subsec:bce_regularization}

Với cả nhãn dương và nhãn âm, huấn luyện trở thành bài toán phân loại nhị phân. Hàm Binary Cross-Entropy được tối thiểu hóa cùng số hạng điều chuẩn $L_2$ (weight decay của bộ tối ưu):
\begin{equation}
    \mathcal{L} = -\sum_{(u, i) \in \mathcal{Y} \cup \mathcal{Y}^-} \left[ y_{ui} \log \hat{y}_{ui} + (1 - y_{ui}) \log (1 - \hat{y}_{ui}) \right] + \lambda_\Theta \|\Theta\|_2^2
    \label{eq:bce_loss_full}
\end{equation}
trong đó $\Theta$ là toàn bộ tham số có thể học (Embedding ID, Embedding thuộc tính, ma trận chiếu, tháp MLP, lớp đầu ra) và $\lambda_\Theta \in \{0;\, 10^{-6}\}$ được tinh chỉnh. BCE được tính trên logit (\texttt{BCEWithLogitsLoss}) để ổn định số học; bộ tối ưu là Adam \parencite{kingma2014adam}, kích thước lô 512.

\textbf{Dừng sớm (Early Stopping) và huấn luyện lại:} sau mỗi epoch, mạng được chấm NDCG@10 theo Full Ranking trên tập xác thực; trạng thái tốt nhất được giữ lại, và huấn luyện dừng khi chỉ số không cải thiện sau 5 epoch liên tiếp. Số epoch tốt nhất $e^*$ được ghi lại để dùng khi huấn luyện lại: trước khi dự đoán cho giai đoạn kiểm thử, mô hình được huấn luyện lại \emph{từ đầu} trên mọi cặp trước mốc kiểm thử đúng $e^*$ epoch -- lúc này không còn tập xác thực để dừng, và tập kiểm thử không bao giờ được dùng để dừng (Thuật toán~\ref{alg:train}, mục~\ref{sec:protocol}).

\begin{algorithm}[htbp]
\caption{Huấn luyện có dừng sớm và huấn luyện lại đúng số epoch}
\label{alg:train}
\begin{algorithmic}[1]
\Require cấu hình $\theta$; seed $s$; dữ liệu trước mốc xác thực $\mathcal{D}_{\tau_{val}}$ và tập xác thực $V$; dữ liệu trước mốc kiểm thử $\mathcal{D}_{\tau_{test}}$; $E_{max} = 20$, kiên nhẫn $P = 5$
\Ensure mô hình $f^*$ để suy diễn sau mốc $\tau_{test}$ và số epoch $e^*$
\Function{HuấnLuyện}{$\mathcal{D}, E, V$}
    \State khởi tạo $f$ theo $\theta$ và seed $s$; $\textit{tốt\_nhất} \gets -\infty$; $e^* \gets 0$; $\textit{chờ} \gets 0$
    \For{$e = 1, \dots, E$}
        \State lấy lại mẫu âm (Thuật toán~\ref{alg:negative}); xáo trộn
        \For{mỗi lô 512 mẫu}
            \State cập nhật $f$ bằng Adam theo BCE trên logit (Công thức~\eqref{eq:bce_loss_full})
        \EndFor
        \If{$V \neq \emptyset$}
            \State $n \gets$ NDCG@10 của $f$ trên $V$ (Full Ranking, Thuật toán~\ref{alg:fullranking})
            \If{$n > \textit{tốt\_nhất}$}
                \State $\textit{tốt\_nhất} \gets n$; $e^* \gets e$; lưu trọng số của $f$; $\textit{chờ} \gets 0$
            \Else
                \State $\textit{chờ} \gets \textit{chờ} + 1$; \textbf{nếu} $\textit{chờ} \ge P$ \textbf{thì} dừng vòng lặp
            \EndIf
        \Else
            \State $e^* \gets e$ \Comment{huấn luyện lại: chạy đủ $E$ epoch}
        \EndIf
    \EndFor
    \State \textbf{nếu} $V \neq \emptyset$ \textbf{thì} nạp lại trọng số tốt nhất
    \State \Return $f, e^*$
\EndFunction
\State $(\cdot, e^*) \gets$ \Call{HuấnLuyện}{$\mathcal{D}_{\tau_{val}}, E_{max}, V$} \Comment{chọn số epoch}
\State $(f^*, \cdot) \gets$ \Call{HuấnLuyện}{$\mathcal{D}_{\tau_{test}}, e^*, \emptyset$} \Comment{huấn luyện lại từ đầu}
\State lưu trọng số $f^*$ \Comment{bước ``lưu mô hình'' của Hình~\ref{fig:system_architecture}}
\State \Return $f^*, e^*$
\end{algorithmic}
\end{algorithm}

\textbf{Nhật ký tinh chỉnh:} mỗi cấu hình ghi một dòng gồm tham số, mã băm cấu hình, phiên bản mã nguồn (commit) và \emph{mã băm nội dung mã nguồn} (gồm toàn bộ thư mục mã của mô hình, dữ liệu, đánh giá và tệp cấu hình), epoch tốt nhất, thời gian và chỉ số trên tập xác thực. Mã băm nội dung cho phép kiểm tra rằng mọi cấu hình được chạy với cùng một mã nguồn, kể cả khi chạy trước lúc lưu phiên bản, và rằng đánh giá cuối dùng đúng mã nguồn đã dùng khi tinh chỉnh.

Tập kiểm thử của mẫu kiểm định chỉ được mở \textbf{đúng một lần}, ở đánh giá cuối; tập kiểm thử của mẫu phát triển không được dựng ở bản cuối của quy trình.
    \item \textbf{Đăng ký trước (pre-registration).} Danh sách mô hình, lưới tinh chỉnh, số seed, độ đo chính, họ so sánh, tiêu chí kết luận và các hạn chế đã biết được ghi vào kế hoạch thực nghiệm (03/10/2026) \emph{trước} khi có kết quả tinh chỉnh của bất kỳ mô hình có đặc trưng nào, rồi được lưu vào hệ thống quản lý phiên bản. Bản cuối của kế hoạch (10/10/2026) đổi bốn điểm theo góp ý của giảng viên hướng dẫn và quyết định của nhóm -- một sản phẩm là một mẫu thiết kế, bỏ sản phẩm chưa có người mua khỏi tập ứng viên, ghi dữ liệu đã lọc ra tệp kiểm bằng mã MD5, và đánh giá cuối trên mẫu kiểm định mới (khối 2). Các điểm này được quyết định sau khi đã xem kết quả trên khối 1, nên khối 1 không còn dùng để kết luận (mục~\ref{sec:v1_history}). Bản cuối được lưu phiên bản sau khi tinh chỉnh lại trên dữ liệu mới của mẫu phát triển và trước khi dựng tập kiểm thử của mẫu kiểm định; độ đo chính, họ so sánh và tiêu chí kết luận giữ nguyên như bản 03/10/2026. Mô hình đề xuất (NeuMF-F) được cố định từ trước, không chọn lại theo kết quả. Mọi thay đổi sau đó được ghi vào mục ``lệch kế hoạch'' kèm thời điểm. Kế hoạch kiểm chứng trên Amazon Beauty được đăng ký riêng theo cùng cách, trước khi chạy mô hình nào trên bộ này. Mô hình lai truyền thống Hybrid-CBCF, thêm theo góp ý của giảng viên hướng dẫn sau khi đánh giá cuối đã xong, có một kế hoạch đăng ký bổ sung riêng, lưu phiên bản trước khi chấm: cách kết hợp chọn trên tập xác thực của mẫu phát triển (Thuật toán~\ref{alg:hybrid}), tập kiểm thử của mỗi bộ dữ liệu được mở thêm một lần cho mô hình này, và hai so sánh bổ sung được hiệu chỉnh Holm riêng, không làm thay đổi họ 10 so sánh dưới đây (mục~\ref{subsec:bosung_results}).
    \item \textbf{Khoá tập kiểm thử.} Chương trình đánh giá cuối từ chối chạy nếu kế hoạch chưa được lưu phiên bản, thiếu lý do chạy, mã nguồn có thay đổi chưa lưu hoặc có tệp mã chưa được theo dõi, tinh chỉnh chưa đủ mô hình, mã băm nội dung mã nguồn khác lúc tinh chỉnh, hoặc seed đó đã từng mở tập kiểm thử. Mỗi seed ghi một dòng nhật ký truy cập (thời gian, phiên bản mã nguồn, mã băm cấu hình, danh sách mô hình, lý do) \emph{trước} khi chấm.
    \item \textbf{Cùng ngân sách tinh chỉnh} cho mọi mô hình có học (6 cấu hình, mục~\ref{subsec:hyperparameters_setup}).
    \item \textbf{Huấn luyện lại trước khi chấm.} Với mỗi seed trong 5 seed (42, 2024, 2025, 2026, 3407) và mỗi mạng nơ-ron: huấn luyện trên dữ liệu trước 01/07/2020 của mẫu kiểm định, dừng sớm theo NDCG@10 trên tập xác thực của chính mẫu này để chọn số epoch $e^*$; huấn luyện lại từ đầu trên mọi cặp trước 29/07/2020 đúng $e^*$ epoch (không còn tập xác thực để dừng, và tập kiểm thử không được dùng để dừng); rồi chấm tập kiểm thử (Thuật toán~\ref{alg:train}). BPR-MF và các mô hình không phải mạng nơ-ron được khớp trên mọi cặp trước 29/07/2020; LateFusion-F trộn đúng GMF-F và MLP-F đã huấn luyện lại ở cùng seed, với $w$ chọn trên mẫu phát triển.
    \item \textbf{Kiểm định thống kê và tiêu chí kết luận cố định.} Với mỗi khách được đánh giá, NDCG@10 được lấy trung bình qua 5 seed (mô hình tất định cho cùng giá trị ở mọi seed), rồi so sánh từng cặp mô hình bằng kiểm định Wilcoxon signed-rank theo người dùng \parencite{wilcoxon1945} và khoảng tin cậy 95\% của hiệu trung bình bằng bootstrap ghép cặp (10.000 lần lấy mẫu lại) \parencite{efron1993bootstrap}; giá trị $p$ được hiệu chỉnh Holm \parencite{holm1979} trên một họ \textbf{10 so sánh} đăng ký trước: NeuMF-F so với NeuMF (đặc trưng có giúp mô hình lai không), với GMF-F và MLP-F (hợp nhất so với từng nhánh), với LateFusion-F (hợp nhất sớm so với muộn), với BPR-MF, ItemKNN, UserKNN, MostPopular-Recent, Content; và NeuMF so với BPR-MF (mô hình lai chỉ dùng ID so với MF). Chỉ viết ``A tốt hơn B'' khi đồng thời: $p_{Holm} < 0{,}05$, khoảng tin cậy không chứa 0, và chênh lệch tương đối NDCG@10 $\ge 5\%$; ngược lại kết luận ``không khác biệt có ý nghĩa''.
\end{enumerate}

Thuật toán~\ref{alg:fullranking} là cách mọi mô hình được chấm trên một giai đoạn -- cùng tập ứng viên, cùng quy tắc phá hoà, nên chênh lệch giữa hai mô hình chỉ đến từ điểm số của chúng. Hạng của từng sản phẩm đúng được lưu lại theo từng khách, nên chỉ số ở một $K$ bất kỳ tính lại được mà \emph{không} phải chấm lại tập kiểm thử; các chỉ số ở $K = 15$ trong Chương~\ref{chap:thucnghiem} được tính theo cách này, và cách tính lại cho đúng các số @5, @10, @20 của lần đánh giá cuối. Thuật toán~\ref{alg:latefusion} chọn trọng số trộn của LateFusion-F chỉ trên tập xác thực của mẫu phát triển; trọng số này (\VCfg{LateF}{params}) được giữ nguyên ở đánh giá cuối.

\begin{algorithm}[htbp]
\caption{Đánh giá Full Ranking một mô hình trên một giai đoạn}
\label{alg:fullranking}
\begin{algorithmic}[1]
\Require hàm điểm $s(u, i)$ của mô hình; tập khách được đánh giá $\mathcal{U}_{eval}$; với mỗi khách $u$: tập ứng viên $\mathcal{C}_u$ và tập sản phẩm đúng $T_u$ (Công thức~\eqref{eq:v2_cand}, \eqref{eq:v2_target}); seed phá hoà $s_0 = 2026$; các giá trị $K$
\Ensure chỉ số trung bình theo khách; hạng của từng sản phẩm đúng
\For{mỗi khách $u \in \mathcal{U}_{eval}$}
    \State tính $s(u, i)$ cho mọi $i \in \mathcal{C}_u$ theo lô \Comment{mạng nơ-ron: logit}
    \State sắp $\mathcal{C}_u$ theo $s(u, i)$ giảm dần; bằng điểm thì theo khoá $h(u, i, s_0)$ \Comment{cùng khoá cho mọi mô hình}
    \State $R_u \gets$ hạng (tính từ 1) của từng $i \in T_u$ trong danh sách đã sắp; lưu $R_u$
    \For{mỗi $K$}
        \State tính HR@$K(u)$, Precision@$K(u)$, Recall@$K(u)$, NDCG@$K(u)$ từ $R_u$ (Công thức~\eqref{eq:hr_k}--\eqref{eq:ndcg_k})
    \EndFor
\EndFor
\State \Return trung bình từng chỉ số trên $\mathcal{U}_{eval}$ và các $R_u$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[htbp]
\caption{Chọn trọng số $w$ của LateFusion-F trên tập xác thực của mẫu phát triển}
\label{alg:latefusion}
\begin{algorithmic}[1]
\Require GMF-F và MLP-F tốt nhất (cấu hình đã chọn khi tinh chỉnh); tập xác thực $V$ của mẫu phát triển; lưới $W = \{0;\ 0{,}1;\ \dots;\ 1\}$
\Ensure trọng số $w^*$
\For{mỗi $w \in W$}
    \For{mỗi khách $u$ của $V$}
        \State $a \gets \mathrm{minmax}\big(s^{\text{GMF-F}}(u, \mathcal{C}_u)\big)$; \quad $b \gets \mathrm{minmax}\big(s^{\text{MLP-F}}(u, \mathcal{C}_u)\big)$ \Comment{Công thức~\eqref{eq:late_fusion_mfdnn}}
        \State điểm trộn của $u$: $w\,a + (1 - w)\,b$
    \EndFor
    \State $N(w) \gets$ NDCG@10 của điểm trộn trên $V$ (Thuật toán~\ref{alg:fullranking})
\EndFor
\State \Return $w^* = \arg\max_{w \in W} N(w)$ \Comment{cố định từ đây, không chọn lại trên tập kiểm thử}
\end{algorithmic}
\end{algorithm}

Thuật toán~\ref{alg:hybrid} chọn cách kết hợp của mô hình lai truyền thống Hybrid-CBCF (mục~\ref{sec:baselines}), cũng chỉ trên tập xác thực của mẫu phát triển. Trọng số của cách kết hợp có trọng số được học, không chọn trên lưới, nên số NDCG@10 dùng để so với các cách bỏ phiếu được tính bằng khớp chéo theo khách: trọng số dùng để chấm một khách luôn được học trên những khách khác. Cách kết hợp và trọng số được chọn giữ nguyên ở đánh giá trên mẫu kiểm định và trên Amazon Beauty.

\begin{algorithm}[htbp]
\caption{Chọn cách kết hợp và học trọng số của mô hình lai truyền thống Hybrid-CBCF trên tập xác thực của mẫu phát triển}
\label{alg:hybrid}
\begin{algorithmic}[1]
\Require Content, UserKNN, ItemKNN với cấu hình đã chọn khi tinh chỉnh, khớp trên dữ liệu trước mốc xác thực; tập xác thực $V$; $m = 50$ sản phẩm không đúng mỗi sản phẩm đúng (rút bằng seed cố định)
\Ensure cách kết hợp $F^*$; trọng số $w^*$ nếu $F^*$ là tổng có trọng số
\For{mỗi khách $u$ của $V$ và mỗi $i \in \mathcal{C}_u$}
    \State $x_{ui} \gets \big(\mathrm{minmax}(s^{CB}(u, \mathcal{C}_u))_i,\ \mathrm{minmax}(s^{UserKNN}(u, \mathcal{C}_u))_i,\ \mathrm{minmax}(s^{ItemKNN}(u, \mathcal{C}_u))_i\big)$
\EndFor
\For{$F \in \{\text{trung bình}, \max, \min\}$} \Comment{bỏ phiếu}
    \State $N(F) \gets$ NDCG@10 trên $V$ của điểm $F(x_{ui})$ (Thuật toán~\ref{alg:fullranking})
\EndFor
\State chia khách của $V$ thành hai phần $P_0, P_1$ theo chỉ số chẵn, lẻ \Comment{khớp chéo}
\For{$k \in \{0, 1\}$}
    \State $D \gets \emptyset$
    \For{mỗi khách $u \notin P_k$ và mỗi sản phẩm đúng $p$ của $u$}
        \State rút $m$ sản phẩm không đúng $j \in \mathcal{C}_u$; thêm $(x_{up} - x_{uj},\ 1)$ và $(x_{uj} - x_{up},\ 0)$ vào $D$
    \EndFor
    \State $w_k \gets$ hồi quy logistic không hệ số chặn trên $D$ \Comment{$\max \sum \ln \sigma(w \cdot (x_{up} - x_{uj}))$}
    \State điểm của mỗi khách $u \in P_k$: $w_k \cdot x_{ui}$
\EndFor
\State $N(\text{trọng số}) \gets$ NDCG@10 trên $V$ của điểm khớp chéo
\State $F^* \gets \arg\max_F N(F)$; nếu $F^*$ là tổng có trọng số thì $w^* \gets$ hồi quy logistic trên các cặp của mọi khách của $V$
\State \Return $F^*, w^*$ \Comment{cố định từ đây, không chọn lại trên tập kiểm thử}
\end{algorithmic}
\end{algorithm}

Ablation (một seed) được báo cáo kèm khoảng tin cậy nhưng không kiểm định và không dùng để kết luận chính; độ phủ, số epoch và thời gian huấn luyện chỉ mang tính mô tả. Mọi lần chạy đánh giá cuối được ghi vào nhật ký truy cập tập kiểm thử; nếu phải chạy lại một seed (ví dụ do lỗi máy), lần chạy lại được đánh dấu riêng và ghi vào mục lệch kế hoạch -- điều này đã xảy ra một lần, với seed 42 (mục~\ref{sec:exp_config}). Trên Amazon Beauty, quy trình giữ nguyên từ bước đăng ký trước trở đi (Hình~\ref{fig:eval_protocol}); khác biệt duy nhất là không có bước tinh chỉnh và không có mẫu khách hàng độc lập (mục~\ref{subsec:amazon_data}).

\section{Suy diễn Top-\texorpdfstring{$K$}{K} và xử lý khách hàng mới}
\label{sec:inference}

Bước cuối của đường ống (Hình~\ref{fig:system_architecture}) là suy diễn: với một khách hàng và một số $K$, hệ thống trả về $K$ sản phẩm nên gợi ý. Thuật toán~\ref{alg:inference} mô tả bước này đúng như cách hệ thống demo (mục~\ref{sec:demo_design}) đang chạy. Với khách đã có trong dữ liệu đã học, đây chính là các bước chấm điểm và sắp xếp của Thuật toán~\ref{alg:fullranking}, chỉ khác là không có sản phẩm đúng để so sánh. Vector của mọi khách và mọi sản phẩm tại mốc $\tau$ được tính một lần khi nạp mô hình, nên mỗi yêu cầu gợi ý chỉ còn một lượt suy diễn thuận trên các sản phẩm ứng viên của khách.

\begin{algorithm}[htbp]
\caption{Suy diễn Top-$K$ cho một khách hàng, có xử lý khách hàng mới}
\label{alg:inference}
\begin{algorithmic}[1]
\Require mã khách $c$; số gợi ý $K$; mô hình đã lưu $f^*$ (NeuMF-F) cùng dữ liệu đã học trước mốc $\tau$; danh sách phổ biến $L_{pop}$ (sản phẩm xếp theo số khách đã mua trước $\tau$)
\Ensure $K$ sản phẩm gợi ý
\If{$c$ không có trong dữ liệu đã học} \Comment{khách mới, hoặc quá ít lần mua để qua lọc $k$-core}
    \State \Return $K$ sản phẩm đầu của $L_{pop}$, gắn nhãn ``gợi ý phổ biến''
\EndIf
\State $u \gets$ mã nội bộ của $c$
\State $\mathcal{C}_u \gets \{i : i$ có dữ liệu huấn luyện trước $\tau\} \setminus \mathcal{I}_u(\tau)$ \Comment{bỏ sản phẩm khách đã mua}
\State lấy $p^G_u, p^M_u$ và $q^G_i, q^M_i$ với mọi $i \in \mathcal{C}_u$ \Comment{đã tính sẵn tại $\tau$ khi nạp mô hình}
\For{mỗi lô sản phẩm $i \in \mathcal{C}_u$}
    \State $s_{ui} \gets h^\top \big[\,p^G_u \odot q^G_i;\ \mathrm{MLP}([p^M_u;\ q^M_i])\,\big]$ \Comment{logit; $\sigma$ không đổi thứ tự}
\EndFor
\State sắp $\mathcal{C}_u$ theo $s_{ui}$ giảm dần (bằng điểm: khoá cố định)
\State \Return $K$ sản phẩm đầu, kèm tên, loại và ảnh
\end{algorithmic}
\end{algorithm}

\paragraph{Khách hàng mới (khởi đầu lạnh của người dùng).} Mô hình chỉ có Embedding ID và đặc trưng thời gian cho những khách đã có trong dữ liệu đã học. Một khách mới -- chưa mua gì, hoặc mua quá ít để qua bước lọc 10-core -- là tình huống khởi đầu lạnh của người dùng (mục~\ref{subsec:thachthuc_sparsity}): mô hình đề xuất không có thông tin về sở thích của khách này. Hệ thống vì vậy dùng một quy tắc đơn giản: \textbf{khách mới được gợi ý các sản phẩm phổ biến nhất} -- danh sách Most Popular tính trên dữ liệu trước mốc, cũng là một mô hình đối chứng được đánh giá ở Chương~\ref{chap:thucnghiem}; \textbf{khi khách đã có lịch sử mua trong dữ liệu đã học, hệ thống dùng mô hình đề xuất NeuMF-F}. Nhờ vậy hệ thống luôn trả về một danh sách gợi ý, không để trống. Đánh giá định lượng của đề tài chỉ đo nhóm khách đã có lịch sử -- mọi khách được đánh giá đều có dữ liệu huấn luyện trước mốc (mục~\ref{subsec:split_sampling}) -- nên chất lượng gợi ý cho khách mới không được đo riêng.

Hai nguyên tắc thiết kế quan trọng nhất: \textit{(i)} tách suy diễn khỏi huấn luyện, nên thời gian phản hồi không phụ thuộc chi phí huấn luyện và tầng 2+3 có thể đóng gói thành một container độc lập; \textit{(ii)} demo \emph{tự kiểm tra} rằng dữ liệu dựng lại khớp với lần chạy đã huấn luyện -- nếu lệch (ví dụ sai tệp dữ liệu hoặc sai ngưỡng $k$), máy chủ báo lỗi thay vì nạp trọng số với ánh xạ ID sai. Tập ứng viên và sản phẩm đích của mỗi khách được lấy thẳng từ dữ liệu dựng lại của đánh giá cuối. Bộ kiểm thử tự động của demo kiểm tra rằng hạng của từng sản phẩm đích và các chỉ số theo người dùng mà demo tính lại khớp tệp kết quả theo người dùng của Chương~\ref{chap:thucnghiem}, lịch sử hiển thị đúng là dữ liệu mô hình đã học, và demo từ chối nạp kết quả được tạo từ dữ liệu khác. Nếu đánh giá cuối được chạy trên GPU, logit tính lại trên CPU có thể làm vài cặp sản phẩm gần như hoà điểm đổi chỗ ở hạng rất sâu (trên 100), không ảnh hưởng chỉ số @K.

\subsection{Thiết kế API suy diễn}
\label{subsec:demo_api}

\begin{table}[htbp]
    \centering
    \caption{Các endpoint chính của API suy diễn}
    \label{tab:demo_api}
    \begin{tabularx}{\textwidth}{|l|X|}
        \hline
        \textbf{Endpoint} & \textbf{Chức năng} \\
        \hline
        \texttt{GET /api/context} & Thông tin bộ dữ liệu, lần chạy đang dùng, số người dùng/sản phẩm, số ứng viên, các $K$ hợp lệ, mô hình khả dụng. \\
        \hline
        \texttt{GET /api/users/search}, \texttt{/random} & Tìm hoặc chọn ngẫu nhiên một khách hàng có sản phẩm đích, lọc theo nhóm ít/trung bình/nhiều giao dịch. \\
        \hline
        \texttt{GET /api/users/\{id\}/history} & Lịch sử mua mà mô hình đã học (mọi cặp trước mốc kiểm thử), sắp theo thời gian; khách mới: lịch sử rỗng, đánh dấu là khách mới. \\
        \hline
        \texttt{GET /api/users/\{id\}/recommend} & Top-$K$ của một mô hình, hạng của từng sản phẩm đích, số ứng viên, HR@$K$, NDCG@$K$, Recall@$K$ của khách đó ($K \in \{5, 10, 15, 20\}$). Khách mới: danh sách Most Popular gắn nhãn ``gợi ý phổ biến''; khách có lịch sử nhưng không mua gì trong giai đoạn kiểm thử: gợi ý của mô hình, không có chỉ số. \\
        \hline
        \texttt{GET /api/users/\{id\}/neighbors} & Top-$K$ khách tương đồng nhất theo UserKNN: độ tương đồng, số sản phẩm mua chung, láng giềng nào đã mua sản phẩm đích của khách. \\
        \hline
        \texttt{GET /api/dashboard} & Số liệu tổng hợp đọc từ tệp kết quả đã chạy, kèm tên tệp nguồn. \\
        \hline
        \texttt{POST /api/reload} & Xoá cache, nạp lần chạy mới nhất mà không cần khởi động lại. \\
        \hline
    \end{tabularx}
\end{table}

Luồng xử lý của chức năng gợi ý Top-$K$ gồm 5 bước, theo đúng Thuật toán~\ref{alg:inference} và dùng lại đúng các hàm của quy trình đánh giá:
\begin{enumerate}
    \item Tra định danh khách hàng để lấy mã nội bộ. Khách không có trong dữ liệu đã học (chưa có lịch sử mua, hoặc quá ít lần mua để qua lọc $k$-core) là tình huống khởi đầu lạnh của người dùng: API trả về danh sách sản phẩm phổ biến (Most Popular) gắn nhãn ``gợi ý phổ biến'', không báo lỗi và không để trống (mục~\ref{sec:inference}); các bước dưới đây chỉ áp dụng cho khách đã có lịch sử.
    \item Lấy tập ứng viên của khách: các sản phẩm có dữ liệu huấn luyện trước mốc, trừ những sản phẩm khách đã mua -- đúng tập ứng viên của đánh giá Full Ranking (mục~\ref{subsec:split_sampling}).
    \item Tính điểm cho toàn bộ tập ứng viên theo lô (vector hoá), dùng logit với mô hình nơ-ron.
    \item Sắp xếp giảm dần với cùng quy tắc phá hoà tất định như lúc đánh giá, lấy $K$ phần tử đầu.
    \item Ánh xạ ngược mã nội bộ về mã mẫu thiết kế (\texttt{product\_code}) và biến thể đại diện (\texttt{article\_id}, dùng để lấy ảnh), kèm tên, loại sản phẩm và số màu của mẫu, trả về Tầng 1.
\end{enumerate}

\subsection{Lựa chọn mô hình triển khai và yêu cầu phi chức năng}
\label{subsec:demo_model_choice}

Demo cho phép chọn và so sánh song song hai mô hình bất kỳ trong số 13 mô hình của đánh giá cuối (mặc định NeuMF-F và NeuMF chỉ dùng ID), để người xem tự đối chiếu với kết luận của Chương~\ref{chap:thucnghiem} -- ví dụ đặt danh sách của NeuMF-F cạnh danh sách của NeuMF chỉ dùng ID trên cùng một khách; demo không khẳng định mô hình nào ``tốt nhất'' ngoài những gì kiểm định thống kê cho phép. Demo nạp đúng các trọng số của đánh giá cuối (seed 42) trên mẫu kiểm định của H\&M, hoặc trên Amazon Beauty: lịch sử hiển thị là đúng dữ liệu mô hình đã học, và sản phẩm đích là các món khách mua lần đầu trong giai đoạn kiểm thử -- nên số liệu trên demo là hiển thị lại kết quả của Chương~\ref{chap:thucnghiem}, không phải một lần đánh giá mới trên tập kiểm thử. Màn hình Admin còn hiện 10 khách tương đồng nhất theo UserKNN để minh họa cách lọc cộng tác dựa trên láng giềng chọn ra gợi ý (mục~\ref{subsec:memory_model_based}). Về yêu cầu phi chức năng, vector của mọi khách và sản phẩm tại mốc dự đoán được tính một lần khi nạp, nên mỗi yêu cầu gợi ý chỉ là một lượt suy diễn thuận trên khoảng \VData{Btest}{cand} ứng viên -- đủ nhanh để phục vụ trực tuyến trên CPU.
