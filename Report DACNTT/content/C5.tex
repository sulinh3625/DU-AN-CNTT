\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
\label{chap:ketluan}

\section{Kết luận}
\label{sec:ketluan}

Đối chiếu với các mục tiêu ở mục~\ref{subsec:muctieu}, đề tài đạt được các kết quả sau:
\begin{enumerate}
    \item \textbf{Cơ sở lý thuyết:} Hệ thống hóa lý thuyết về lọc cộng tác trên phản hồi ẩn, nhân tử hóa ma trận, BPR, GMF, MLP, mô hình lai NeuMF, hai cách hợp nhất Early Fusion và Late Fusion, mô hình lai dựa trên đặc trưng và khởi đầu lạnh của sản phẩm \parencite{he2017ncf, koren2009matrix, rendle2009bpr, rendle2010fm, volkovs2017dropoutnet}, cùng các nghiên cứu về phương pháp đánh giá hệ gợi ý \parencite{rendle2020ncfmf, dacrema2019progress, krichene2020sampled}; đối chiếu thiết lập thực nghiệm của ba nghiên cứu nền tảng (mục~\ref{subsec:foundation_studies}).
    \item \textbf{Quy trình dữ liệu tái lập được, không rò rỉ thông tin tương lai:} lấy hai mẫu khách hàng không giao nhau từ 31,8 triệu giao dịch H\&M; coi mỗi mẫu thiết kế (\texttt{product\_code}) là một sản phẩm; lọc $k$-core chỉ trên dữ liệu trước mốc kiểm thử; ghi dữ liệu đã lọc ra tệp kiểm bằng mã MD5 để mọi mô hình dùng đúng cùng dữ liệu; đặc trưng sản phẩm, khách hàng và thời gian được tính theo thời điểm; lấy mẫu âm chỉ trong sản phẩm đã ra mắt; các điều kiện này được kiểm chứng bằng bộ kiểm thử tự động (mục~\ref{sec:hm_pipeline}). Các bước chính được trình bày thành mã giả (Thuật toán~\ref{alg:kcore}--\ref{alg:inference}).
    \item \textbf{Mô hình đề xuất NeuMF-F:} giữ cấu trúc hai nhánh GMF và MLP hợp nhất sớm của NeuMF, biểu diễn mỗi khách hàng và sản phẩm bằng Embedding ID cộng phép chiếu của đặc trưng, với cơ chế bỏ ngẫu nhiên ID sản phẩm khi huấn luyện là một siêu tham số được tinh chỉnh (mục~\ref{sec:model_design}); cùng các mô hình đối chiếu trên cùng khung (GMF-F, MLP-F, LateFusion-F, NeuMF chỉ dùng ID) và bảy baseline được tinh chỉnh cùng ngân sách; các mô hình đối chứng được xếp thành ba nhóm -- mô hình đơn lẻ, học máy truyền thống, chuyên biệt cho gợi ý (mục~\ref{sec:baselines}).
    \item \textbf{Đánh giá không thiên lệch:} tinh chỉnh chỉ trên tập xác thực của mẫu phát triển; đánh giá cuối một lần trên mẫu khách hàng độc lập theo kế hoạch đăng ký trước, có khoá tập kiểm thử và nhật ký truy cập; 5 seed; kiểm định Wilcoxon theo người dùng kèm bootstrap và hiệu chỉnh Holm trên họ 10 so sánh (mục~\ref{sec:protocol}); chỉ số ở $K = 5, 10, 15, 20$.
    \item \textbf{Kết quả chính trên H\&M:} Trên tập kiểm thử của mẫu kiểm định, NeuMF-F đạt NDCG@10 = \VRes{NeuMFF}{NDCG10} $\pm$ \VSd{NeuMFF}{NDCG10} (xếp \VRank{NeuMFF}{NDCG10} trong 14 mô hình, sau LateFusion-F với \VRes{LateF}{NDCG10}); NeuMF chỉ dùng ID đạt \VRes{NeuMF}{NDCG10}, BPR-MF \VRes{BPR}{NDCG10}, UserKNN \VRes{UKNN}{NDCG10}, MostPopular-Recent \VRes{RecPop}{NDCG10}. \VSig{all}{summary} Bổ sung đặc trưng là nguồn cải thiện chính -- NeuMF-F so với NeuMF chỉ dùng ID chênh \VSig{1}{rel} -- trong đó đặc trưng thời gian đóng góp nhiều nhất (bỏ đi, NDCG@10 thay đổi \VAbl{AblTime}{rel}). Hợp nhất sớm hai nhánh vượt nhánh MLP-F đứng riêng (\VSig{3}{rel}) nhưng không khác biệt có ý nghĩa với nhánh GMF-F đứng riêng (\VSig{2}{rel}) và với hợp nhất muộn LateFusion-F (\VSig{4}{rel}). Bốn mô hình có đặc trưng giữ nguyên thứ tự và vị trí dẫn đầu ở mọi $K = 5, 10, 15, 20$. Kết quả được báo cáo đầy đủ theo đúng kế hoạch đã đăng ký (Chương~\ref{chap:thucnghiem}).
    \item \textbf{Kiểm chứng trên Amazon Beauty:} chạy lại toàn bộ đánh giá cuối với cấu hình chọn trên H\&M. Kết luận về đặc trưng còn đúng: NeuMF-F hơn NeuMF chỉ dùng ID \ASig{1}{rel} và hơn có ý nghĩa cả năm baseline. Kết luận về cách hợp nhất không còn đúng hoàn toàn: NeuMF-F không khác biệt có ý nghĩa với GMF-F lẫn MLP-F, và LateFusion-F tốt hơn NeuMF-F có ý nghĩa (NeuMF-F thấp hơn \ASig{4}{rel}). So sánh với BPR-MF trên Amazon không dùng để kết luận vì cấu hình của BPR-MF chọn trên H\&M không hợp với dữ liệu này (mục~\ref{sec:amazon_results}).
    \item \textbf{Phân tích bổ sung với mô hình lai truyền thống:} Hybrid-CBCF kết hợp Content, UserKNN và ItemKNN theo mô hình lai trong luận án của \textcite{holinh2023multisource}, với trọng số học theo cặp trên tập xác thực của mẫu phát triển (Thuật toán~\ref{alg:hybrid}); mô hình được đăng ký bổ sung và chấm một lần trên tập kiểm thử của mỗi bộ dữ liệu. NeuMF-F so với Hybrid-CBCF chênh \VSig{B1}{rel} trên H\&M (\VSig{B1}{phrase}) và \ASig{B1}{rel} trên Amazon Beauty (\ASig{B1}{phrase}). Phân tích này không thay đổi các kết luận đã đăng ký (mục~\ref{subsec:bosung_results}).
    \item \textbf{Khả năng triển khai:} Hệ thống demo FastAPI chỉ suy diễn, dùng đúng các mô hình của đánh giá cuối trên H\&M và Amazon Beauty, tự kiểm tra khớp dữ liệu với lần chạy đã huấn luyện; khách hàng mới chưa có lịch sử được gợi ý sản phẩm phổ biến, khách đã có lịch sử nhận gợi ý của mô hình đề xuất, nên hệ thống luôn có gợi ý (mục~\ref{sec:inference}, \ref{sec:demo_design}).
\end{enumerate}

\section{Hạn chế của đề tài}
\label{sec:hanche}

\begin{itemize}
    \item \textbf{Quy mô dữ liệu:} Hai mẫu khoảng 500.000 giao dịch (mỗi mẫu khoảng 1,5\% khách hàng), không phải toàn bộ 31,8 triệu giao dịch, vì đánh giá Full Ranking trên toàn bộ danh mục vượt khả năng của phần cứng sẵn có (máy tính cá nhân, RAM 16 GB).
    \item \textbf{Ngưỡng $k$-core = 10:} chọn vì giới hạn bộ nhớ thay vì ngưỡng thường gặp $k=5$; dữ liệu sau lọc thiên về khách mua nhiều, nên kết quả chưa đại diện cho khách ít giao dịch.
    \item \textbf{Phạm vi sản phẩm:} sản phẩm chưa có người mua trước mốc dự đoán -- trong đó có sản phẩm vừa ra mắt -- không thuộc tập ứng viên, nên kết quả không nói về khả năng gợi ý sản phẩm mới; gợi ý ở mức mẫu thiết kế, không ở mức màu. Thuộc tính khách hàng không có mốc thời gian nên được coi là ít thay đổi.
    \item \textbf{Khách hàng mới:} khách chưa có lịch sử mua được gợi ý danh sách phổ biến; đề tài không đánh giá định lượng chất lượng gợi ý cho nhóm khách này.
    \item \textbf{Kiểm chứng trên bộ dữ liệu thứ hai còn hạn chế:} kết luận trên H\&M đúng cho khách hàng khác của cùng cửa hàng trong giai đoạn 07--09/2020. Phần kiểm chứng trên Amazon Beauty dùng lại cấu hình chọn trên H\&M (một số mô hình, nhất là BPR-MF, có thể bị thiệt), không có mẫu khách hàng độc lập, coi mỗi đánh giá là một lần mua, và dùng nhóm giá tại thời điểm thu thập dữ liệu (năm 2014) chứ không phải giá tại ngày của từng đánh giá.
    \item \textbf{Mô hình lai truyền thống được thêm sau đánh giá cuối:} Hybrid-CBCF được quyết định thêm khi kết quả kiểm thử của 14 mô hình -- kể cả ba thành phần của nó -- đã biết. Thiết kế lấy từ luận án của giảng viên hướng dẫn và cách kết hợp chọn trên tập xác thực, nhưng kết quả của nó chỉ là phân tích bổ sung; trọng số học trên H\&M được dùng nguyên cho Amazon Beauty, và thành phần Content trên máy chạy bổ sung lệch nhẹ so với lần đánh giá cuối do tệp đặc trưng văn bản được dựng lại (mục~\ref{subsec:bosung_results}).
    \item \textbf{Ngân sách tinh chỉnh:} mỗi mô hình có học chỉ được thử 6 cấu hình; vài cấu hình được chọn nằm ở biên lưới (mục~\ref{sec:pham_vi_thuc_nghiem}), nên một số mô hình có thể còn tốt hơn nếu mở rộng lưới.
    \item \textbf{Phạm vi đặc trưng và mô hình:} chưa dùng ảnh sản phẩm, giá (trên H\&M), chuỗi hành vi; mô tả văn bản chỉ được biểu diễn bằng TF-IDF và SVD; ablation chỉ chạy một seed.
    \item \textbf{Lịch sử phát triển:} tập kiểm thử của mẫu phát triển đã được xem nhiều lần ở giai đoạn phát triển ban đầu, và tập kiểm thử của khối 1 đã được mở ở bản đầu của kế hoạch đánh giá trước khi kế hoạch được sửa thành bản cuối (mục~\ref{sec:v1_history}); hai lần đánh giá này vì vậy chỉ được dùng như lịch sử phát triển. Thiết kế bản cuối có tham khảo những gì rút ra từ hai giai đoạn đó -- nhưng không dùng bất kỳ dữ liệu nào của mẫu kiểm định.
\end{itemize}

\section{Hướng phát triển}
\label{sec:huongphattrien}

\begin{enumerate}
    \item \textbf{Đặc trưng đa phương thức:} biểu diễn mô tả sản phẩm bằng mô hình ngôn ngữ (ví dụ BERT) và ảnh sản phẩm bằng mạng tích chập, hoặc dùng Transformer để hợp nhất nhiều view như hướng của \textcite{holinh2023multisource}, với cùng quy trình đánh giá không thiên lệch để đo đúng lợi ích so với vector TF-IDF hiện tại.
    \item \textbf{Gợi ý sản phẩm mới:} đưa lại sản phẩm chưa có người mua vào tập ứng viên, với một giả định về danh mục được nêu rõ, để đo riêng khả năng xử lý khởi đầu lạnh của sản phẩm -- điều bản đầu của kế hoạch đánh giá đã thử nhưng không đo được rõ ở quy mô mẫu hiện tại (mục~\ref{sec:v1_history}).
    \item \textbf{Khách hàng mới:} thay danh sách phổ biến chung bằng danh sách phổ biến gần đây hoặc theo nhóm khách (tuổi, khu vực mua sắm) và đánh giá riêng trên nhóm khách chưa có lịch sử.
    \item \textbf{Hợp nhất muộn có trọng số học được:} kết quả trên Amazon cho thấy hợp nhất muộn có thể tốt hơn hợp nhất sớm; bước tiếp theo là thay trọng số cố định của Late Fusion bằng trọng số học được theo nhóm khách hàng (ví dụ ít/nhiều giao dịch) hoặc theo loại sản phẩm (ví dụ bán chạy/ít bán).
    \item \textbf{Mở rộng quy mô và kiểm chứng chéo:} chạy trên mẫu lớn hơn với $k=5$ hoặc trên toàn bộ H\&M bằng Full Ranking theo lô trên GPU, lặp lại trên khung thời gian khác; trên Amazon, tinh chỉnh riêng cho bộ dữ liệu này để tách ảnh hưởng của cấu hình khỏi bản thân mô hình. Quy trình hiện tại tách phần đọc dữ liệu thành adapter riêng nên chỉ cần bổ sung một adapter mới cho mỗi bộ dữ liệu.
    \item \textbf{Mô hình chuỗi, đồ thị và tự mã hóa biến phân:} thử SASRec \parencite{kang2018self}, LightGCN \parencite{he2020lightgcn} hoặc Mult-VAE \parencite{liang2018multvae} trên cùng giao thức, với baseline được tinh chỉnh cùng ngân sách.
    \item \textbf{Triển khai:} kiểm tra đầy đủ bản đóng gói Docker của demo; với danh mục lớn, dùng kiến trúc hai tháp và tìm kiếm lân cận gần đúng \parencite{covington2016youtube} thay vì chấm điểm toàn bộ danh mục bằng MLP.
\end{enumerate}

\section{Triển khai thực tế giao diện thực nghiệm}
\label{sec:huongthietke_giaodien}

Thiết kế hệ thống demo (mục~\ref{sec:demo_design}) đã được cài đặt thành hệ thống chạy được: chọn bộ dữ liệu H\&M hoặc Amazon Beauty, nạp đúng trọng số của đánh giá cuối để suy diễn trực tiếp, tự đối chiếu dữ liệu với lần chạy trước khi nạp, tính lại các chỉ số theo người dùng khớp với kết quả của Chương~\ref{chap:thucnghiem} ở $K = 5, 10, 15, 20$, và minh họa ``Top-$K$ khách tương đồng'' bằng UserKNN. Khách đã có lịch sử nhận gợi ý của mô hình được chọn; khách chưa có trong dữ liệu đã học nhận danh sách sản phẩm phổ biến kèm nhãn ``gợi ý phổ biến'' thay vì thông báo lỗi (Thuật toán~\ref{alg:inference}). Tệp Dockerfile đã có nhưng chưa được kiểm tra đầy đủ; đây là công việc tiếp theo nêu ở mục~\ref{sec:huongphattrien}.
