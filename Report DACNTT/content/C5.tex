\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
\label{chap:ketluan}

\section{Kết luận}
\label{sec:ketluan}

Đối chiếu với các mục tiêu ở mục~\ref{subsec:muctieu}, đề tài đạt được các kết quả sau:
\begin{enumerate}
    \item \textbf{Cơ sở lý thuyết:} Hệ thống hóa lý thuyết về lọc cộng tác trên phản hồi ẩn, nhân tử hóa ma trận, BPR, GMF, MLP và mô hình lai NeuMF \parencite{he2017ncf, koren2009matrix, rendle2009bpr}, cùng các hướng hiện đại (mục~\ref{sec:modern_trends}) và các nghiên cứu về đánh giá công bằng \parencite{rendle2020ncfmf, dacrema2019progress, krichene2020sampled} để xác định vị trí và giới hạn của hướng tiếp cận.
    \item \textbf{Quy trình dữ liệu tái lập được, không rò rỉ:} Lấy mẫu hm500k theo khách hàng từ 31,8 triệu giao dịch H\&M (tất định, khoảng 70 giây, 0,5 GB RAM); gộp cặp User--Item trước khi lọc $k$-core; chia theo một mốc thời gian chung để mô hình không thấy tương lai; kiểm tra không giao nhau và kiểm thử tự động trên dữ liệu thật (mục~\ref{sec:hm_pipeline}).
    \item \textbf{Cài đặt mô hình:} GMF, MLP, NeuMF-Scratch và NeuMF-Pretrained bằng PyTorch \parencite{paszke2019pytorch}, lấy mẫu âm động chỉ từ tập huấn luyện, pre-training với nhánh GMF và MLP có thể khác số chiều; xác định rõ NeuMF là mô hình lai MF + DNN theo kiểu Early Fusion ở mức biểu diễn (mục~\ref{subsec:early_late_fusion}) và chỉ ra rằng biến thể ``nối Embedding ngay ở đầu vào'' cài đặt ban đầu trùng kiến trúc với MLP đơn lẻ (mục~\ref{subsec:early_fusion_design}); cài đặt thêm Late Fusion trộn điểm của chính GMF và MLP, và ItemKNN, UserKNN dạng ma trận thưa top-$k$ chạy được với khoảng 10.000 sản phẩm.
    \item \textbf{Đánh giá không thiên lệch:} Tinh chỉnh cùng ngân sách (6 cấu hình/mô hình) chỉ trên validation, khoá tập test với nhật ký truy cập, đăng ký trước tiêu chí kết luận, 3 seed, kiểm định Wilcoxon theo người dùng kèm bootstrap và hiệu chỉnh Holm (mục~\ref{sec:protocol}); phân tích bóc tách thành phần gồm bỏ từng nhánh, tiền huấn luyện, và ablation số chiều Embedding, số tầng MLP, tỉ lệ mẫu âm theo đề cương (mục~\ref{subsec:ablation}).
    \item \textbf{Kết quả chính:} Trên tập kiểm thử, BPR-MF đạt NDCG@10 trung bình cao nhất (\Res{BPR}{NDCG10}); NeuMF-Pretrained (\Res{Pre}{NDCG10}) ngang Most Popular (\Res{MostPop}{NDCG10}); \Sig{all}{nbetter}/\Sig{all}{n} so sánh đạt tiêu chí ``tốt hơn''. NeuMF không vượt baseline MF đã tinh chỉnh và không vượt các nhánh thành phần của chính nó; pre-training không mang lại lợi ích đo được (Chương~\ref{chap:thucnghiem}). Kết quả âm này được báo cáo đúng như kế hoạch đã đăng ký và nhất quán với \textcite{rendle2020ncfmf}.
    \item \textbf{Mở rộng (sau khi xem test):} so sánh NeuMF (Early Fusion) với Late Fusion của chính GMF và MLP, và đặt các mô hình học cạnh ItemKNN, UserKNN, trong một họ 7 so sánh riêng (mục~\ref{sec:extension_results}). \Esig{all}{summary}
    \item \textbf{Khả năng triển khai:} Hệ thống demo FastAPI chỉ suy diễn, dùng đúng các mô hình của đánh giá cuối (số trên demo khớp Chương~\ref{chap:thucnghiem}), tự kiểm tra khớp dữ liệu với lần chạy đã huấn luyện, minh họa Top-$K$ khách tương đồng theo UserKNN; độ trễ gợi ý top-10 được đo thực tế trên CPU (mục~\ref{subsec:latency}).
\end{enumerate}

\section{Hạn chế của đề tài}
\label{sec:hanche}

\begin{itemize}
    \item \textbf{Quy mô dữ liệu:} Chỉ dùng mẫu hm500k (khoảng 1,6\% khách hàng, 500.269 giao dịch), không phải toàn bộ 31,8 triệu giao dịch, vì đánh giá Full Ranking trên toàn bộ catalog vượt khả năng của phần cứng (GPU 4 GB, RAM 16 GB).
    \item \textbf{Ngưỡng $k$-core = 10:} Chọn vì giới hạn bộ nhớ thay vì ngưỡng chuẩn $k=5$; dữ liệu sau lọc thiên về khách mua nhiều, nên kết quả chưa đại diện cho khách ít giao dịch.
    \item \textbf{Dữ liệu thưa:} 201.801 cặp huấn luyện (209.212 khi gộp tập xác thực để huấn luyện lại); các mô hình nơ-ron đạt đỉnh sau 1--9 epoch. Khi huấn luyện lại trên tập huấn luyện $\cup$ xác thực, số epoch được lấy từ lần huấn luyện chỉ trên tập huấn luyện vì không còn tập xác thực để dừng sớm.
    \item \textbf{Lệch kế hoạch sau khi xem test:} Bước huấn luyện lại trên tập huấn luyện $\cup$ xác thực được bổ sung sau lần chấm test đầu tiên, bốn mô hình mở rộng được thêm sau lần chấm thứ hai, và toàn bộ chuỗi đánh giá được chạy lại một lần trên phiên bản mã nguồn cuối (mục~\ref{sec:protocol}). Mọi thay đổi đều được đăng ký trước khi chạy và ghi trong nhật ký truy cập, nhưng số liệu của Chương~\ref{chap:thucnghiem} vẫn là kết quả sau khi tập kiểm thử đã được xem.
    \item \textbf{Chỉ 3 seed:} Đủ cho kiểm định theo người dùng nhưng không đủ để kiểm định theo seed; độ lệch chuẩn giữa các seed của mô hình nơ-ron cùng cỡ với chênh lệch giữa các mô hình.
    \item \textbf{Phạm vi mô hình:} Chỉ lọc cộng tác thuần dựa trên ID -- chưa dùng thuộc tính sản phẩm, ảnh, văn bản, thông tin khách hàng; chưa mô hình hóa chuỗi hành vi. iALS và mô hình trộn điểm iALS + NeuMF đã bị loại khỏi phạm vi dù cho kết quả validation tốt hơn NeuMF; Late Fusion và hai baseline láng giềng chỉ được đánh giá ở phần mở rộng sau khi đã xem test; Late Fusion chỉ dùng trọng số cố định chọn trên validation (chưa thử trọng số học được); ablation số chiều Embedding, số tầng MLP và tỉ lệ mẫu âm chỉ làm trên tập xác thực với một seed.
    \item \textbf{Cold-start tuyệt đối} (người dùng/sản phẩm chưa có giao dịch nào) nằm ngoài khả năng của mô hình; demo chỉ xử lý bằng gợi ý theo luật, được gắn nhãn rõ.
    \item \textbf{Một bộ dữ liệu:} Kết luận chưa được kiểm chứng chéo trên dữ liệu có mật độ và quy mô catalog khác.
\end{itemize}

\section{Hướng phát triển}
\label{sec:huongphattrien}

\begin{enumerate}
    \item \textbf{Kiểm chứng phần mở rộng trên dữ liệu chưa từng xem.} Late Fusion và baseline láng giềng mới chỉ được đánh giá sau khi đã xem tập kiểm thử; cần đăng ký trước và đánh giá lại trên một bộ dữ liệu giao dịch thứ hai, cùng với iALS được tune \parencite{rendle2022ials} và phép trộn điểm iALS + NeuMF-Scratch -- hai mô hình đều vượt NeuMF trên validation -- để trả lời liệu NeuMF có giá trị như một thành phần bổ sung cho MF hay không. Có thể thay trọng số cố định của Late Fusion bằng trọng số học được theo nhóm người dùng (ví dụ ít/nhiều tương tác), như cách kết hợp bằng mạng nơ-ron trong \textcite{holinh2023multisource}.
    \item \textbf{Mở rộng quy mô:} Chạy trên toàn bộ H\&M (hoặc mẫu lớn hơn với $k=5$) bằng đánh giá Full Ranking theo lô trên GPU lớn hơn, hoặc Full Ranking trên một mẫu con người dùng cố định.
    \item \textbf{Tăng số seed} (từ 6 trở lên) để có thể kiểm định cả theo seed, và lặp lại trên một bộ dữ liệu giao dịch thứ hai; quy trình hiện tại tách phần đọc dữ liệu thành adapter riêng nên chỉ cần bổ sung một adapter mới.
    \item \textbf{Đặc trưng nội dung và đa phương thức:} Tích hợp thuộc tính sản phẩm, ảnh và mô tả văn bản để hỗ trợ sản phẩm ít tương tác và Cold-start -- ví dụ Early Fusion vector ẩn của MF với vector BERT của mô tả sản phẩm H\&M, hoặc dùng Transformer để hợp nhất nhiều view như hướng của \textcite{holinh2023multisource} -- với cùng quy trình đánh giá không thiên lệch để đo đúng lợi ích.
    \item \textbf{Mô hình chuỗi, đồ thị và tự mã hóa biến phân:} Thử SASRec \parencite{kang2018self}, LightGCN \parencite{he2020lightgcn} hoặc Mult-VAE \parencite{liang2018multvae} trên cùng giao thức chia theo thời gian, với baseline được tune cùng ngân sách.
    \item \textbf{Triển khai:} Kiểm tra đầy đủ bản đóng gói Docker của demo và dùng tìm kiếm lân cận gần đúng cho MF khi catalog lớn.
\end{enumerate}

\section{Triển khai thực tế giao diện thực nghiệm}
\label{sec:huongthietke_giaodien}

Thiết kế hệ thống demo (mục~\ref{sec:demo_design}) đã được cài đặt thành hệ thống chạy được: nạp đúng trọng số của đánh giá cuối để suy diễn trực tiếp, tự đối chiếu dữ liệu với lần chạy trước khi nạp, tính lại HR@K, NDCG@K, Recall@K theo người dùng khớp với kết quả của Chương~\ref{chap:thucnghiem}, và minh họa ``Top-$K$ khách tương đồng'' bằng UserKNN (Hình~\ref{fig:demo_neighbors}). Trang Dashboard đọc trực tiếp các tệp kết quả cuối (bảng kết quả, kiểm định, phân tầng, độ trễ). Gợi ý cho khách hàng mới dùng luật đơn giản và được gắn nhãn rõ là không phải mô hình NeuMF. Tệp Dockerfile đã có nhưng chưa được kiểm tra đầy đủ; đây là công việc tiếp theo nêu ở mục~\ref{sec:huongphattrien}.
