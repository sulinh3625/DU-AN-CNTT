\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
\label{chap:ketluan}

\section{Kết luận}
\label{sec:ketluan}

Đối chiếu với các mục tiêu ở mục~\ref{subsec:muctieu}, đề tài đạt được các kết quả sau:
\begin{enumerate}
    \item \textbf{Cơ sở lý thuyết:} Hệ thống hóa lý thuyết về lọc cộng tác trên phản hồi ẩn, nhân tử hóa ma trận, BPR, GMF, MLP và mô hình lai NeuMF \parencite{he2017ncf, koren2009matrix, rendle2009bpr}, cùng các hướng hiện đại (mục~\ref{sec:modern_trends}) và các nghiên cứu về đánh giá công bằng \parencite{rendle2020ncfmf, dacrema2019progress, krichene2020sampled} để xác định vị trí và giới hạn của hướng tiếp cận.
    \item \textbf{Quy trình dữ liệu tái lập được, không rò rỉ:} Lấy mẫu hm500k theo khách hàng từ 31,8 triệu giao dịch H\&M (tất định, khoảng 70 giây, 0,5 GB RAM); gộp cặp User--Item trước khi lọc $k$-core; chia theo một mốc thời gian chung để mô hình không thấy tương lai; kiểm tra không giao nhau và kiểm thử tự động trên dữ liệu thật (mục~\ref{sec:hm_pipeline}).
    \item \textbf{Cài đặt mô hình:} GMF, MLP, NeuMF-Scratch và NeuMF-Pretrained bằng PyTorch \parencite{paszke2019pytorch}, lấy mẫu âm động chỉ từ tập huấn luyện, pre-training với nhánh GMF và MLP có thể khác số chiều; xác định rõ NeuMF là mô hình lai MF + DNN theo kiểu Early Fusion ở mức biểu diễn (mục~\ref{subsec:early_late_fusion}) và chỉ ra rằng biến thể ``nối Embedding ngay ở đầu vào'' cài đặt ban đầu trùng kiến trúc với MLP đơn lẻ (mục~\ref{subsec:early_fusion_design}).
    \item \textbf{Đánh giá không thiên lệch:} Tinh chỉnh cùng ngân sách (6 cấu hình/mô hình) chỉ trên validation, khoá tập test với nhật ký truy cập, đăng ký trước tiêu chí kết luận, 3 seed, kiểm định Wilcoxon theo người dùng kèm bootstrap và hiệu chỉnh Holm (mục~\ref{sec:protocol}).
    \item \textbf{Kết quả chính:} Trên tập kiểm thử, BPR-MF đạt NDCG@10 trung bình cao nhất (\Res{BPR}{NDCG10}); NeuMF-Pretrained (\Res{Pre}{NDCG10}) ngang Most Popular (\Res{MostPop}{NDCG10}); \Sig{all}{nbetter}/\Sig{all}{n} so sánh đạt tiêu chí ``tốt hơn''. NeuMF không vượt baseline MF đã tinh chỉnh và không vượt các nhánh thành phần của chính nó; pre-training không mang lại lợi ích đo được (Chương~\ref{chap:thucnghiem}). Kết quả âm này được báo cáo đúng như kế hoạch đã đăng ký và nhất quán với \textcite{rendle2020ncfmf}.
    \item \textbf{Khả năng triển khai:} Hệ thống demo FastAPI chỉ suy diễn, tự kiểm tra khớp dữ liệu với lần chạy đã huấn luyện, và độ trễ gợi ý top-10 được đo thực tế trên CPU (mục~\ref{subsec:latency}).
\end{enumerate}

\section{Hạn chế của đề tài}
\label{sec:hanche}

\begin{itemize}
    \item \textbf{Quy mô dữ liệu:} Chỉ dùng mẫu hm500k (khoảng 1,6\% khách hàng, 500.269 giao dịch), không phải toàn bộ 31,8 triệu giao dịch, vì đánh giá Full Ranking trên toàn bộ catalog vượt khả năng của phần cứng (GPU 4 GB, RAM 16 GB).
    \item \textbf{Ngưỡng $k$-core = 10:} Chọn vì giới hạn bộ nhớ thay vì ngưỡng chuẩn $k=5$; dữ liệu sau lọc thiên về khách mua nhiều, nên kết quả chưa đại diện cho khách ít giao dịch.
    \item \textbf{Dữ liệu thưa:} 201.801 cặp huấn luyện (209.212 khi gộp tập xác thực để huấn luyện lại); các mô hình nơ-ron đạt đỉnh sau 1--9 epoch. Khi huấn luyện lại trên tập huấn luyện $\cup$ xác thực, số epoch được lấy từ lần huấn luyện chỉ trên tập huấn luyện vì không còn tập xác thực để dừng sớm.
    \item \textbf{Lệch kế hoạch sau khi xem test:} Bước huấn luyện lại trên tập huấn luyện $\cup$ xác thực được bổ sung sau lần chấm test đầu tiên (mục~\ref{sec:protocol}). Kết luận thống kê không đổi giữa hai lần chấm, nhưng số liệu của Chương~\ref{chap:thucnghiem} là kết quả sau khi đã xem test một lần.
    \item \textbf{Chỉ 3 seed:} Đủ cho kiểm định theo người dùng nhưng không đủ để kiểm định theo seed; độ lệch chuẩn giữa các seed của mô hình nơ-ron cùng cỡ với chênh lệch giữa các mô hình.
    \item \textbf{Phạm vi mô hình:} Chỉ lọc cộng tác thuần dựa trên ID -- chưa dùng thuộc tính sản phẩm, ảnh, văn bản, thông tin khách hàng; chưa mô hình hóa chuỗi hành vi. Baseline mạnh ItemKNN không chạy được do giới hạn bộ nhớ; iALS và mô hình trộn điểm iALS + NeuMF đã bị loại khỏi phạm vi dù cho kết quả validation tốt hơn NeuMF; Late Fusion thuần MF + DNN (trộn điểm GMF và MLP) chưa được thực hiện, nên đề tài chưa so sánh được Early Fusion với Late Fusion; số tầng của tháp MLP chưa được khảo sát.
    \item \textbf{Cold-start tuyệt đối} (người dùng/sản phẩm chưa có giao dịch nào) nằm ngoài khả năng của mô hình; demo chỉ xử lý bằng gợi ý theo luật, được gắn nhãn rõ.
    \item \textbf{Một bộ dữ liệu:} Kết luận chưa được kiểm chứng chéo trên dữ liệu có mật độ và quy mô catalog khác.
\end{itemize}

\section{Hướng phát triển}
\label{sec:huongphattrien}

\begin{enumerate}
    \item \textbf{So sánh Early Fusion với Late Fusion và đưa iALS vào đánh giá cuối.} Thực hiện Late Fusion thuần MF + DNN với cùng thành phần như NeuMF (trộn điểm GMF và MLP huấn luyện riêng, trọng số chọn trên tập xác thực) để so trực tiếp với NeuMF; đồng thời chấm test cho iALS được tune \parencite{rendle2022ials} và phép trộn điểm iALS + NeuMF-Scratch -- hai mô hình đều vượt NeuMF trên validation -- sau khi đăng ký trước, đặc biệt để trả lời liệu NeuMF có giá trị như một thành phần bổ sung cho MF hay không.
    \item \textbf{Mở rộng quy mô:} Chạy trên toàn bộ H\&M (hoặc mẫu lớn hơn với $k=5$) bằng đánh giá Full Ranking theo lô trên GPU lớn hơn, hoặc Full Ranking trên một mẫu con người dùng cố định.
    \item \textbf{Tăng số seed} (từ 6 trở lên) để có thể kiểm định cả theo seed, và lặp lại trên một bộ dữ liệu giao dịch thứ hai; quy trình hiện tại tách phần đọc dữ liệu thành adapter riêng nên chỉ cần bổ sung một adapter mới.
    \item \textbf{Đặc trưng nội dung và đa phương thức:} Tích hợp thuộc tính sản phẩm, ảnh và mô tả văn bản để hỗ trợ sản phẩm ít tương tác và Cold-start -- ví dụ Early Fusion vector ẩn của MF với vector BERT của mô tả sản phẩm H\&M, hoặc dùng Transformer để hợp nhất nhiều view như hướng của \textcite{holinh2023multisource} -- với cùng quy trình đánh giá không thiên lệch để đo đúng lợi ích.
    \item \textbf{Mô hình chuỗi và đồ thị:} Thử SASRec \parencite{kang2018self} hoặc LightGCN \parencite{he2020lightgcn} trên cùng giao thức chia theo thời gian, với baseline được tune cùng ngân sách.
    \item \textbf{Triển khai:} Đóng gói demo bằng Docker, trỏ demo tới đúng các checkpoint của đánh giá cuối, và dùng tìm kiếm lân cận gần đúng cho MF khi catalog lớn.
\end{enumerate}

\section{Triển khai thực tế giao diện thực nghiệm}
\label{sec:huongthietke_giaodien}

Thiết kế hệ thống demo (mục~\ref{sec:demo_design}) đã được cài đặt thành hệ thống chạy được: nạp lại trọng số đã huấn luyện để suy diễn trực tiếp, tự đối chiếu dữ liệu với lần chạy trước khi nạp, và tính lại Hit@K/NDCG@K theo người dùng khớp với kết quả thực nghiệm. Gợi ý cho khách hàng mới dùng luật đơn giản và được gắn nhãn rõ là không phải mô hình NeuMF. Tệp Dockerfile đã có nhưng chưa được kiểm tra đầy đủ; đây là công việc tiếp theo nêu ở mục~\ref{sec:huongphattrien}.
