\chapter{TỔNG QUAN VỀ ĐỀ TÀI}
\label{chap:tongquan}

\section{Bối cảnh nghiên cứu và tính cấp thiết của đề tài}
\label{sec:boicanh}

\subsection{Quan sát ban đầu từ dữ liệu và động lực chọn đề tài}
\label{subsec:thuctrang}

Động lực trực tiếp của đề tài xuất phát từ việc khảo sát bộ dữ liệu H\&M Personalized Fashion Recommendations \parencite{hm2022kaggle} trước khi bắt tay vào xây dựng mô hình: tệp giao dịch gốc ghi nhận 31.788.324 dòng mua hàng của 1.362.281 khách hàng trên 104.547 sản phẩm thời trang (mục~\ref{subsec:hm_description}), nhưng ngay cả sau khi lọc $k$-core, ma trận tương tác vẫn chỉ có mật độ khoảng 0,03\% (Bảng~\ref{tab:kcore_hm_full}) -- mỗi khách hàng chỉ mua một phần cực nhỏ trong catalog. Cấu trúc dữ liệu ``nhiều User -- nhiều Item -- cực thưa'' này khác đáng kể so với kịch bản phim ảnh (MovieLens) hay mạng ảnh (Pinterest) mà phần lớn tài liệu về NCF/NeuMF gốc \parencite{he2017ncf} sử dụng để kiểm chứng -- đặt ra câu hỏi cụ thể mà nhóm muốn trả lời qua đề tài: một kiến trúc lai tuyến tính--phi tuyến như NeuMF có còn phát huy ưu thế trên dữ liệu giao dịch thời trang thực tế, cực thưa và lệch mạnh về một nhóm nhỏ sản phẩm phổ biến hay không.

Ở góc độ ứng dụng, việc cá nhân hóa gợi ý sản phẩm trong bối cảnh thương mại điện tử vẫn giữ nguyên giá trị đã được ghi nhận trong tài liệu học thuật -- giảm chi phí tìm kiếm của khách hàng và hỗ trợ dự báo nhu cầu tiêu thụ cho doanh nghiệp \parencite{schafer2001commerce, linden2003amazon} -- nhưng phần đóng góp cụ thể của đề tài này không nằm ở việc lặp lại luận điểm đó, mà ở việc \emph{đo lường bằng số liệu thực} xem một kiến trúc học sâu cụ thể (NeuMF) hoạt động ra sao trên đúng loại dữ liệu giao dịch có cấu trúc như trên.

\subsection{Vì sao Nhân tử hóa ma trận tuyến tính chưa đủ cho dữ liệu này}
\label{subsec:hanche_cf}

Cách tiếp cận đơn giản nhất cho bài toán trên là Nhân tử hóa ma trận (Matrix Factorization -- MF): mỗi User $u$ và Item $i$ được gán một vector ẩn $p_u, q_i \in \mathbb{R}^d$, và mức độ tương tác được dự đoán bằng tích vô hướng của hai vector này \parencite{koren2009matrix}:
\begin{equation}
    \hat{y}_{ui} = p_u^T q_i = \sum_{f=1}^{d} p_{uf} q_{if}
    \label{eq:mf_intro}
\end{equation}
Vấn đề nhóm nhận thấy khi cân nhắc áp dụng trực tiếp công thức này lên H\&M là: tích vô hướng cộng dồn các chiều đặc trưng theo cùng một trọng số cố định (ngầm định bằng 1 cho mọi chiều), trong khi với một ma trận tương tác cực thưa (mật độ 0,28\% trên mẫu thực nghiệm sau lọc $k$-core, Bảng~\ref{tab:kcore_hm}) và độ lệch phổ biến rõ rệt, mỗi vector ẩn được học từ rất ít tương tác và khả năng cao là chỉ một vài chiều đặc trưng thực sự mang tính phân biệt, còn lại là nhiễu. He và cộng sự \parencite{he2017ncf} chỉ ra vấn đề này ở mức tổng quát hơn: tích vô hướng là một phép kết hợp tuyến tính, không có cơ chế học trọng số riêng cho từng chiều hay mô hình hóa tương tác phi tuyến giữa các chiều đặc trưng -- đây là cơ sở lý thuyết trực tiếp cho lựa chọn kiến trúc ở mục tiếp theo.

\subsection{Lựa chọn NeuMF: kết hợp thay vì thay thế MF}
\label{subsec:sumenh_neumf}

Thay vì từ bỏ hoàn toàn cấu trúc tích vô hướng của MF (vốn đã được kiểm chứng hiệu quả và dễ diễn giải), NeuMF -- do He và cộng sự đề xuất năm 2017 \parencite{he2017ncf} -- giữ lại một nhánh tương đương MF dưới dạng mạng nơ-ron (Generalized Matrix Factorization, GMF) để không đánh mất khả năng học tương tác tuyến tính đã biết là hoạt động tốt, đồng thời bổ sung song song một nhánh Multi-Layer Perceptron (MLP) để mạng có cơ hội tự học thêm các tương tác phi tuyến mà GMF bỏ lỡ. Đây là lý do cụ thể đề tài chọn NeuMF làm đối tượng nghiên cứu chính thay vì một kiến trúc học sâu khác: nó cho phép \emph{tách bạch} được đóng góp của thành phần tuyến tính và phi tuyến trong cùng một mô hình (thông qua Ablation Study, mục~\ref{subsec:ablation}) -- điều mà một mạng nơ-ron end-to-end không có cấu trúc hai nhánh tường minh sẽ khó thực hiện tương đương.

\section{Mục tiêu và phạm vi nghiên cứu}
\label{sec:muctieu_phamvi}

\subsection{Mục tiêu nghiên cứu}
\label{subsec:muctieu}

\begin{itemize}
    \item Nghiên cứu cơ sở lý thuyết về Lọc cộng tác, Nhân tử hóa ma trận, Mạng nơ-ron sâu và kiến trúc mô hình lai NeuMF \parencite{he2017ncf, koren2009matrix}.
    \item Xây dựng quy trình xử lý dữ liệu hoàn chỉnh, áp dụng trên bộ dữ liệu H\&M Personalized Fashion Recommendations \parencite{hm2022kaggle} -- biến đổi dữ liệu giao dịch thành ma trận tương tác User--Item dạng tín hiệu ẩn, thực hiện lấy mẫu tiêu cực (Negative Sampling) và phân chia tập dữ liệu theo thời gian sao cho mô hình không nhìn thấy tương lai.
    \item Thiết kế, cài đặt và tinh chỉnh các mô hình GMF, MLP độc lập và mô hình lai NeuMF trên nền tảng PyTorch, với cùng ngân sách tinh chỉnh như các baseline.
    \item Đánh giá thực nghiệm định lượng hiệu năng của NeuMF so với các nhánh thành phần (GMF, MLP) và baseline (Random, Most Popular, BPR-MF) qua các độ đo NDCG@K \parencite{jarvelin2002}, Recall@K, HR@K, Precision@K theo giao thức Full Ranking; thực hiện phân tích bóc tách thành phần (Ablation Study), đánh giá phân tầng theo độ phổ biến sản phẩm và mức độ hoạt động của người dùng, và kiểm định ý nghĩa thống kê theo người dùng qua nhiều lần chạy (3 seed).
    \item So sánh hai cách hợp nhất mô hình MF với mạng nơ-ron sâu -- hợp nhất ở mức biểu diễn (Early Fusion, chính là NeuMF) và hợp nhất ở mức điểm dự đoán (Late Fusion) của cùng hai thành phần GMF và MLP -- và đặt các mô hình học cạnh baseline láng giềng ItemKNN, UserKNN (phần mở rộng, mục~\ref{sec:extension_results}).
\end{itemize}

\textit{Lưu ý về phạm vi:} Dự án tập trung vào nhóm người dùng và sản phẩm đã có đủ lịch sử tương tác để xây dựng biểu diễn Embedding (sau khi lọc $k$-core). Bài toán khởi đầu lạnh (Cold-start) cho người dùng/sản phẩm hoàn toàn mới không nằm trong phạm vi đánh giá thực nghiệm của dự án và được ghi nhận như một hướng phát triển trong tương lai (Chương~\ref{chap:ketluan}).

\subsection{Phạm vi nghiên cứu}
\label{subsec:phamvi}

\begin{itemize}
    \item \textbf{Đối tượng nghiên cứu:} Kiến trúc Neural Collaborative Filtering (GMF, MLP, NeuMF) \parencite{he2017ncf}, hai cách hợp nhất mô hình MF với mạng nơ-ron sâu -- Early Fusion (mức biểu diễn) và Late Fusion (mức điểm dự đoán) \parencite{holinh2023multisource} -- kỹ thuật biểu diễn Embedding và các thuật toán đánh giá hệ thống gợi ý Top-K.
    \item \textbf{Dữ liệu thực nghiệm:} Bộ dữ liệu H\&M Personalized Fashion Recommendations \parencite{hm2022kaggle} (thời trang, catalog lớn và cực thưa), khai thác lịch sử giao dịch mua sắm giữa khách hàng và sản phẩm. Khảo sát $k$-core được chạy trên toàn bộ 31,8 triệu dòng giao dịch; huấn luyện và đánh giá mô hình được thực hiện trên mẫu hm500k -- khoảng 500.000 giao dịch của 21.599 khách hàng được chọn ngẫu nhiên, giữ trọn lịch sử mua 2 năm (lý do ở mục~\ref{subsec:hm_sampling}).
    \item \textbf{Tín hiệu tương tác:} Phản hồi ẩn dạng nhị phân dựa trên lịch sử mua hàng. Mọi mô hình chỉ học từ ma trận tương tác User--Item (lọc cộng tác thuần); không dùng giá, thuộc tính sản phẩm, ảnh hay thông tin khách hàng làm đầu vào.
\end{itemize}

\section{Phương pháp nghiên cứu}
\label{sec:phuongphap}

Để giải quyết các mục tiêu nghiên cứu đã đặt ra (mục~\ref{subsec:muctieu}), đề tài kết hợp bốn nhóm phương pháp bổ trợ lẫn nhau, được triển khai tuần tự theo đúng trình tự trình bày trong các chương tiếp theo:

\begin{enumerate}
    \item \textbf{Phương pháp nghiên cứu lý thuyết (phân tích và tổng hợp tài liệu):} Hệ thống hóa cơ sở lý thuyết về Lọc cộng tác, Nhân tử hóa ma trận và kiến trúc NCF/NeuMF từ các công trình nền tảng \parencite{koren2009matrix, he2017ncf}, đồng thời khảo sát các hướng nghiên cứu hiện đại hơn (mô hình lai dựa trên đặc trưng, gợi ý theo chuỗi và Transformer, Graph-based CF, tự mã hóa biến phân) để xác định chính xác vị trí đóng góp và các giới hạn cố hữu của hướng tiếp cận được lựa chọn (Chương~\ref{chap:lythuyet}).
    \item \textbf{Phương pháp xử lý dữ liệu định lượng:} Xây dựng một pipeline tiền xử lý dữ liệu tường minh, có thể tái lập (reproducible) và kiểm chứng được ở từng bước -- từ lấy mẫu theo khách hàng, gộp giao dịch trùng lặp, lọc $k$-core, đến phân chia theo một mốc thời gian chung kèm kiểm tra không giao nhau bắt buộc -- áp dụng trên bộ dữ liệu H\&M Personalized Fashion Recommendations (Chương~\ref{chap:thietke_trienkhai}).
    \item \textbf{Phương pháp thực nghiệm khoa học (mô hình hóa và huấn luyện):} Cài đặt các mô hình GMF, MLP và NeuMF bằng framework PyTorch \parencite{paszke2019pytorch}, áp dụng chiến lược tiền huấn luyện (Pre-training) và tinh chỉnh (Fine-tuning) của He và cộng sự \parencite{he2017ncf}, dùng thuật toán tối ưu Adam \parencite{kingma2014adam}; siêu tham số của mọi mô hình được tinh chỉnh với cùng ngân sách, chỉ trên tập xác thực.
    \item \textbf{Phương pháp đánh giá định lượng và kiểm chứng thống kê:} Đo lường hiệu năng bằng các độ đo chuẩn (NDCG@K \parencite{jarvelin2002}, Recall@K, HR@K, Precision@K) theo giao thức Full Ranking, dùng Sampled-99 chỉ để đối chiếu \parencite{krichene2020sampled}; khoá tập kiểm thử và đăng ký trước tiêu chí kết luận; kiểm định Wilcoxon theo người dùng kèm khoảng tin cậy bootstrap và hiệu chỉnh Holm \parencite{wilcoxon1945, efron1993bootstrap, holm1979}; bổ sung các chỉ số ngoài độ chính xác (Coverage, Novelty, tỉ lệ sản phẩm phổ biến) để tránh kết luận chỉ dựa trên độ chính xác (Chương~\ref{chap:thucnghiem}).
\end{enumerate}

Điểm xuyên suốt của phương pháp luận là ưu tiên tính có thể tái lập và kiểm chứng được: mọi số liệu trình bày trong báo cáo đều được trích xuất trực tiếp từ kết quả do chính quy trình thực nghiệm sinh ra, thay vì ước lượng hay diễn giải gián tiếp.

\section{Ý nghĩa khoa học và thực tiễn của đề tài}
\label{sec:ynghia}

\subsection{Ý nghĩa khoa học}
\label{subsec:ynghia_khoahoc}

\begin{itemize}
    \item Cung cấp bằng chứng thực nghiệm, có kiểm định thống kê, trên dữ liệu giao dịch thời trang thực tế và cực thưa (H\&M) về việc NeuMF có vượt baseline MF đã tinh chỉnh và các nhánh thành phần của chính nó (GMF, MLP) hay không -- một câu hỏi mà công trình gốc chủ yếu kiểm chứng trên dữ liệu phim ảnh (MovieLens) và mạng xã hội hình ảnh (Pinterest). Kết quả (không vượt) nhất quán với các nghiên cứu đối chứng gần đây \parencite{rendle2020ncfmf, dacrema2019progress}.
    \item Xác định rõ vị trí của NeuMF trong phân loại Early Fusion/Late Fusion (mục~\ref{subsec:early_late_fusion}): NeuMF là hợp nhất nhánh MF và nhánh DNN ở mức biểu diễn, và trên dữ liệu này không vượt có ý nghĩa thống kê so với từng nhánh đứng riêng; phần mở rộng so sánh trực tiếp NeuMF với Late Fusion của chính GMF và MLP (mục~\ref{sec:extension_results}).
    \item Xây dựng một quy trình đánh giá chống thiên lệch có thể tái sử dụng -- chia theo thời gian không rò rỉ tương lai, cùng ngân sách tinh chỉnh, khoá tập test có nhật ký truy cập, kiểm định cặp có hiệu chỉnh đa so sánh -- áp dụng được cho các nghiên cứu hệ gợi ý khác.
\end{itemize}

\subsection{Ý nghĩa thực tiễn}
\label{subsec:ynghia_thucte}

\begin{itemize}
    \item Xây dựng một quy trình tiền xử lý và huấn luyện tách biệt phần đọc dữ liệu (adapter) khỏi logic xử lý cốt lõi, có thể tái sử dụng cho các bộ dữ liệu giao dịch thương mại điện tử khác bằng cách bổ sung một adapter tương ứng.
    \item Đề xuất cách lấy mẫu theo khách hàng, đọc tệp theo lô (khoảng 0,5 GB RAM cho tệp 3,5 GB) và tái lập được (mục~\ref{subsec:hm_sampling}), giúp thực nghiệm trên dữ liệu giao dịch lớn khả thi trên phần cứng phổ thông mà vẫn giữ trọn lịch sử thời gian của từng khách.
    \item Cung cấp hệ thống demo suy diễn (FastAPI) với độ trễ đo thực tế, cho phép so sánh trực quan các mô hình trên từng khách hàng, và kết luận thực tiễn rằng với dữ liệu giao dịch thưa, một mô hình MF đơn giản được tinh chỉnh tốt là lựa chọn hợp lý hơn NeuMF về cả độ chính xác lẫn chi phí.
\end{itemize}

\section{Bố cục của báo cáo dự án}
\label{sec:bocuc}

Báo cáo dự án được trình bày trong 5 chương chính:
\begin{itemize}
    \item \textbf{Chương 1 -- Tổng quan về đề tài:} Giới thiệu bối cảnh, tính cấp thiết, mục tiêu, phạm vi, phương pháp nghiên cứu và ý nghĩa của đề tài.
    \item \textbf{Chương 2 -- Cơ sở lý thuyết và các công trình liên quan:} Trình bày nền tảng lý thuyết về hệ thống gợi ý, Matrix Factorization, GMF, MLP, NeuMF, các chỉ số đánh giá và tổng quan các nghiên cứu tiền nhiệm.
    \item \textbf{Chương 3 -- Phương pháp nghiên cứu và thiết kế mô hình NeuMF:} Mô tả quy trình xử lý dữ liệu H\&M, kiến trúc chi tiết mô hình NeuMF, kỹ thuật Negative Sampling, chiến lược huấn luyện, quy trình tinh chỉnh và đánh giá không thiên lệch, và thiết kế hệ thống giao diện thực nghiệm (demo).
    \item \textbf{Chương 4 -- Thực nghiệm và đánh giá kết quả:} Trình bày kết quả tinh chỉnh, kết quả trên tập kiểm thử, kiểm định thống kê, phân tích Ablation Study, đánh giá phân tầng, đối chiếu Sampled-99, độ trễ suy diễn và phần mở rộng (Late Fusion, baseline láng giềng).
    \item \textbf{Chương 5 -- Kết luận và hướng phát triển:} Tổng kết kết quả đạt được, chỉ ra các hạn chế và đề xuất hướng mở rộng nghiên cứu.
\end{itemize}
